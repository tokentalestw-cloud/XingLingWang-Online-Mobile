function getUnitStars(unit, zone, lane) {
  if (!unit) return 0;
  const c = unit.card || unit;
  let baseStars = Number(c.score ?? c.stars ?? 0) + Number(unit.bonusScore ?? 0);

  // 細胞旅人 (SR-NMS-0026 / SSR-NMS-0026) 效果：X = 場上細胞旅人數量
  if (c.id === "SR-NMS-0026" || c.id === "SSR-NMS-0026" || c.name?.includes("細胞旅人")) {
    let resolvedZone = zone;
    let resolvedLane = lane;
    if (!resolvedZone || resolvedLane === undefined || resolvedLane === -1) {
      for (const testZone of ["player_front", "player_back", "enemy_front", "enemy_back"]) {
        const testIdx = field[testZone].indexOf(unit);
        if (testIdx >= 0) {
          resolvedZone = testZone;
          resolvedLane = testIdx;
          break;
        }
      }
    }
    if (resolvedZone && resolvedLane !== undefined && resolvedLane !== -1 && window.isUnitSilenced(unit, resolvedZone, resolvedLane)) {
      baseStars = 0 + Number(unit.bonusScore ?? 0);
    } else {
      baseStars = window.countCellTravelers() + Number(unit.bonusScore ?? 0);
    }
  }

  // 衰退扣分
  if (unit.decayScoreDebuff) {
    baseStars -= unit.decayScoreDebuff;
  }

  // 冰霜法師 水瓶座 (SSSR-NMS-0060) 減星效果
  if (unit.equipments && (unit.equipments.includes("冰霜法師(-3)") || unit.equipments.some(e => e.includes("冰霜法師(-3)")))) {
    baseStars -= 3;
  }

  // 千面病毒 增星效果
  if (unit.equipments && (unit.equipments.includes("千面病毒") || unit.equipments.some(e => e.includes("千面病毒")))) {
    baseStars += 1;
  }

  // 金衣服的女人 (R-ART-0038) 連動：場上每個橫置藝術品使此卡星星數 +1
  if ((c.id === "R-ART-0038" || c.id === "R-ART-0038-金" || c.name?.includes("金衣服的女人")) && zone) {
    const isPlayerSide = zone.startsWith("player_");
    const prefix = isPlayerSide ? "player_" : "enemy_";
    let tappedArtCount = 0;
    const frontZone = prefix + "front";
    const backZone = prefix + "back";
    [frontZone, backZone].forEach(z => {
      if (field[z]) {
        field[z].forEach(u => {
          if (u && u.tapped) {
            const uc = u.card;
            if (uc && (uc.faction === "藝術品" || uc.deck === "藝術品" || uc.id?.includes("ART"))) {
              tappedArtCount++;
            }
          }
        });
      }
    });
    baseStars += tappedArtCount;
  }

  if (zone && lane !== undefined && field[zone]) {
    // 創世男孩 (ART-0013) 連動：創世老人在右邊時 +3⭐
    if (c.id === "ART-0013" && lane < 4) {
      const rightUnit = field[zone][lane + 1];
      const rightCard = rightUnit && (rightUnit.card || rightUnit);
      if (rightCard && rightCard.id === "ART-0014") {
        baseStars += 3;
      }
    }

    // 盾甲兵 (SSSR-NMG-0072) 連動：使此單位左右邊的單位 +1⭐
    const adjShieldSoldiers = [lane - 1, lane + 1];
    for (const idx of adjShieldSoldiers) {
      if (idx >= 0 && idx < 5) {
        const adj = field[zone][idx];
        if (adj && adj.card && (adj.card.id === "SSSR-NMG-0072" || adj.card.name?.includes("盾甲兵"))) {
          baseStars += 1;
        }
      }
    }

    // 狐狸雜耍者 達亞 (SSSR-NMS-0056) 連動：若左右邊皆有單位，使左邊單位 -3⭐，右邊單位 +3⭐
    // 1. 檢查我方右邊 (lane + 1) 是否是 狐狸雜耍者 達亞，且它的右邊 (lane + 2) 也有單位
    if (lane < 4) {
      const rightUnit = field[zone][lane + 1];
      if (rightUnit && rightUnit.card && (rightUnit.card.id === "SSSR-NMS-0056" || rightUnit.card.name?.includes("狐狸雜耍者"))) {
        if (!window.isUnitSilenced(rightUnit, zone, lane + 1)) {
          // 檢查達亞右邊也有單位
          if (lane < 3 && field[zone][lane + 2] !== null) {
            baseStars -= 3;
          }
        }
      }
    }
    // 2. 檢查我方左邊 (lane - 1) 是否是 狐狸雜耍者 達亞，且它的左邊 (lane - 2) 也有單位
    if (lane > 0) {
      const leftUnit = field[zone][lane - 1];
      if (leftUnit && leftUnit.card && (leftUnit.card.id === "SSSR-NMS-0056" || leftUnit.card.name?.includes("狐狸雜耍者"))) {
        if (!window.isUnitSilenced(leftUnit, zone, lane - 1)) {
          // 檢查達亞左邊也有單位
          if (lane > 1 && field[zone][lane - 2] !== null) {
            baseStars += 3;
          }
        }
      }
    }
  }

  return baseStars;
}

function getUnitAtk(unit, zone, lane, isBeingAttacked = false) {
  if (!unit) return 0;
  let rainbowBonus = 0;
  if (window.XLW_rainbowAtkBonusActive && unit.attacking) {
    rainbowBonus = 1;
  }
  const isShield = window.XLW_isShieldUnit(unit);
  const c = unit.card || unit;
  let baseAtk = 0;
  if (!isShield) {
    baseAtk = Number(c.attack ?? c.atk ?? 0);
    baseAtk = Number.isFinite(baseAtk) ? baseAtk : 0;
  }

  // 烈陽戰士 獅子座 (SSSR-NMS-0058) 動態攻擊力鏡像
  if (c.id === "SSSR-NMS-0058" || c.name?.includes("烈陽戰士")) {
    let resolvedZone = zone;
    let resolvedLane = lane;
    if (!resolvedZone || resolvedLane === undefined || resolvedLane === -1) {
      for (const testZone of ["player_front", "player_back", "enemy_front", "enemy_back"]) {
        const testIdx = field[testZone].indexOf(unit);
        if (testIdx >= 0) {
          resolvedZone = testZone;
          resolvedLane = testIdx;
          break;
        }
      }
    }
    if (resolvedZone && resolvedLane !== undefined && resolvedLane !== -1 && window.isUnitSilenced(unit, resolvedZone, resolvedLane)) {
      baseAtk = 0;
    } else {
      let targetUnit = unit.leoLinkedTargetUnit;
      if (!targetUnit && resolvedZone) {
        const sidePrefix = resolvedZone.startsWith("player_") ? "player_" : "enemy_";
        for (const testZone of [sidePrefix + "front", sidePrefix + "back"]) {
          for (const u of field[testZone]) {
            if (u && u.equipments && u.equipments.includes("烈陽戰士(+2)")) {
              targetUnit = u;
              break;
            }
          }
          if (targetUnit) break;
        }
      }
      if (targetUnit) {
        if (!window.XLW_leoRecursionGuard) {
          window.XLW_leoRecursionGuard = true;
          try {
            let targetZone = null;
            let targetIdx = -1;
            for (const testZone of ["player_front", "player_back", "enemy_front", "enemy_back"]) {
              const idx = field[testZone].indexOf(targetUnit);
              if (idx >= 0) {
                targetZone = testZone;
                targetIdx = idx;
                break;
              }
            }
            if (targetZone && targetIdx >= 0) {
              baseAtk = getUnitAtk(targetUnit, targetZone, targetIdx);
            } else {
              baseAtk = 0;
            }
          } finally {
            window.XLW_leoRecursionGuard = false;
          }
        } else {
          baseAtk = 0;
        }
      } else {
        baseAtk = 0;
      }
    }
  }

  // 細胞旅人 (SR-NMS-0026 / SSR-NMS-0026) 效果：X = 場上細胞旅人數量
  if (c.id === "SR-NMS-0026" || c.id === "SSR-NMS-0026" || c.name?.includes("細胞旅人")) {
    let resolvedZone = zone;
    let resolvedLane = lane;
    if (!resolvedZone || resolvedLane === undefined || resolvedLane === -1) {
      for (const testZone of ["player_front", "player_back", "enemy_front", "enemy_back"]) {
        const testIdx = field[testZone].indexOf(unit);
        if (testIdx >= 0) {
          resolvedZone = testZone;
          resolvedLane = testIdx;
          break;
        }
      }
    }
    if (resolvedZone && resolvedLane !== undefined && resolvedLane !== -1 && window.isUnitSilenced(unit, resolvedZone, resolvedLane)) {
      baseAtk = 0;
    } else {
      baseAtk = window.countCellTravelers();
    }
  }

  // 不醉翁 (VRT-0005) 效果：X = 我方場上酒客單位數量 * 2
  if (c.id === "VRT-0005" || c.name?.includes("不醉翁")) {
    let resolvedZone = zone;
    let resolvedLane = lane;
    if (!resolvedZone || resolvedLane === undefined || resolvedLane === -1) {
      for (const testZone of ["player_front", "player_back", "enemy_front", "enemy_back"]) {
        const testIdx = field[testZone].indexOf(unit);
        if (testIdx >= 0) {
          resolvedZone = testZone;
          resolvedLane = testIdx;
          break;
        }
      }
    }
    if (resolvedZone && resolvedLane !== undefined && resolvedLane !== -1 && window.isUnitSilenced(unit, resolvedZone, resolvedLane)) {
      baseAtk = 0;
    } else {
      let sidePrefix = resolvedZone && resolvedZone.startsWith("player_") ? "player_" : "enemy_";
      let count = 0;
      for (const z of [sidePrefix + "front", sidePrefix + "back"]) {
        for (let idx = 0; idx < 5; idx++) {
          const u = field[z][idx];
          if (u && isDrinkerUnit(u)) {
            count++;
          }
        }
      }
      baseAtk = count * 2;
    }
  }

  // 河童場地減攻效果
  if (zone && zone.includes("front")) {
    const owner = zone.startsWith("player_") ? "player" : "enemy";
    const fieldEl = owner === "player" ? $("playerField") : $("enemyField");
    if (fieldEl && fieldEl.dataset.card) {
      try {
        const fieldCard = JSON.parse(fieldEl.dataset.card);
        if (fieldCard.name.includes("河童")) {
          baseAtk = baseAtk - 1;
        }
      } catch (err) {}
    }
  }

  // 魔抗鐵拳等動態加減攻
  if (unit && unit.atkModifier) {
    baseAtk = baseAtk + Number(unit.atkModifier);
  }
  if (unit && unit.tempAtkModifier) {
    baseAtk = baseAtk + Number(unit.tempAtkModifier);
  }

  // 計算相鄰強化與連動 Buff
  if (zone && lane !== undefined && field[zone]) {
    // 1. 大鵰像 (ART-0002) 相鄰強化：左右邊的單位 +2 攻擊力
    if (lane > 0) {
      const leftUnit = field[zone][lane - 1];
      const leftCard = leftUnit && (leftUnit.card || leftUnit);
      if (leftCard && leftCard.id === "ART-0002") {
        baseAtk += 2;
      }
    }
    if (lane < 4) {
      const rightUnit = field[zone][lane + 1];
      const rightCard = rightUnit && (rightUnit.card || rightUnit);
      if (rightCard && rightCard.id === "ART-0002") {
        baseAtk += 2;
      }
    }

    // 2. 創世老人 (ART-0014) 連動：創世男孩在左邊時 +4 攻擊力
    if (c.id === "ART-0014" && lane > 0) {
      const leftUnit = field[zone][lane - 1];
      const leftCard = leftUnit && (leftUnit.card || leftUnit);
      if (leftCard && leftCard.id === "ART-0013") {
        baseAtk += 4;
      }
    }

    // 3. 彩虹維修員 (SSSR-NMS-0011) 相鄰強化：左右邊的單位 +1 攻擊力
    const adjRepairers = [lane - 1, lane + 1];
    for (const idx of adjRepairers) {
      if (idx >= 0 && idx < 5) {
        const adj = field[zone][idx];
        if (adj && adj.card && (adj.card.id === "SSSR-NMS-0011" || adj.card.name?.includes("彩虹維修員"))) {
          baseAtk += 1;
        }
      }
    }
  }

  // R-ORC-0007 鐵獸人
  if ((c.id === "R-ORC-0007" || c.name?.includes("鐵獸人")) && isBeingAttacked) {
    baseAtk += 2;
  }
  // R-FMS-0005 臭豆腐屍 正前方攻擊降低 2
  if (zone && lane !== undefined && lane >= 0) {
    const sidePrefix = zone.startsWith("player_") ? "player_" : "enemy_";
    const oppPrefix = zone.startsWith("player_") ? "enemy_" : "player_";
    let oppStinky = false;
    
    // Check direct opposite row
    const oppositeRow = zone.includes("front") ? "front" : "back";
    const oppUnit = field[oppPrefix + oppositeRow][lane];
    if (oppUnit && oppUnit.card && (oppUnit.card.id === "R-FMS-0005" || oppUnit.card.name?.includes("臭豆腐屍")) && !window.isUnitSilenced(oppUnit, oppPrefix + oppositeRow, lane)) {
      oppStinky = true;
    }
    
    if (oppStinky) {
      baseAtk = Math.max(0, baseAtk - 2);
    }
  }
  // R-ORC-0033 阿庫瑪的戰錘
  if (unit.equipments && unit.equipments.some(eq => eq.includes("戰錘") || eq.includes("阿庫瑪的戰錘"))) {
    baseAtk += 1;
  }
  // 彩虹幻象喵 (R-CAT-0030) Buff
  if (unit.attacking) {
    const sideIsPlayer = zone && zone.startsWith("player_");
    if (sideIsPlayer && window.XLW_rainbowCatAtkBuffActive) {
      baseAtk += 1;
    } else if (!sideIsPlayer && window.XLW_enemyRainbowCatAtkBuffActive) {
      baseAtk += 1;
    }
  }
  // SR-ORC-0062 戰爭巨象 艾勒粉
  if (c.id === "SR-ORC-0062" || c.name?.includes("艾勒粉") || c.name?.includes("戰爭巨象")) {
    const orcEquipCount = (unit.equipments || []).filter(eqName => {
      return ["帽", "菜刀", "狼牙棒", "弓", "牛排", "斧", "戰錘", "盾牌", "面具"].some(x => eqName.includes(x));
    }).length;
    baseAtk += orcEquipCount * 2;
  }
  // ORC-0002 背刺獸人
  if ((c.id === "ORC-0002" || c.name?.includes("背刺獸人")) && unit.attacking && unit.target) {
    const defUnit = field[unit.target.zone][unit.target.idx];
    if (defUnit && defUnit.tapped) {
      baseAtk += 3;
    }
  }
  // 有打有嘉獎! (ORC-0020) Buff
  const sideIsPlayer = zone && zone.startsWith("player_");
  if (c.faction === "獸人" || c.id?.includes("ORC")) {
    if ((sideIsPlayer && window.XLW_playerOrcAwardActiveUntil >= turn) || (!sideIsPlayer && window.XLW_enemyOrcAwardActiveUntil >= turn)) {
      baseAtk += 1;
    }
  }

  // 3. 禁衛軍獸人 (SSR-ORC-0057) 被攻擊時 +2 攻擊力
  if (c.id === "SSR-ORC-0057" || c.name?.includes("禁衛軍獸人")) {
    if (isBeingAttacked) {
      baseAtk += 2;
    }
  }

  // 突擊獸人 (ORC-0005) 進攻時 +3 攻擊力
  if ((c.id === "ORC-0005" || c.name === "突擊獸人") && unit.attacking) {
    baseAtk += 3;
  }

  if (zone) {
    const sidePrefix = zone.startsWith("player_") ? "player_" : "enemy_";
    const otherZones = [sidePrefix + "front", sidePrefix + "back"];
    let hasDisimaiPassive = false;
    for (const z of otherZones) {
      if (field[z]) {
        field[z].forEach((u, i) => {
          if (u && u !== unit && u.card && (u.card.id === "SSR-ART-0054" || u.card.name?.includes("迪斯麥")) && !window.isUnitSilenced(u, z, i)) {
            // Check if it is the upgraded version (either starts with SSR or has attack >= 4)
            if (u.card.id === "SSR-ART-0054" || Number(u.card.attack) >= 4) {
              hasDisimaiPassive = true;
            }
          }
        });
      }
    }
    if (hasDisimaiPassive && isRogueUnit(unit)) {
      baseAtk += 1;
    }
  }

  // 5. 臨時非裝備狀態魔法效果 (削弱藥水、振奮藥水、睡眠反擊拳)
  if (unit.equipments) {
    if (unit.equipments.includes("削弱藥水")) {
      baseAtk = baseAtk - 3;
    }
    if (unit.equipments.includes("振奮藥水")) {
      baseAtk += 3;
    }
    if (unit.equipments.includes("睡眠反擊拳")) {
      baseAtk += 3;
    }
    if (unit.equipments.includes("酒意魔菇(+3)")) {
      baseAtk += 3;
    }
    if (unit.equipments.includes("惡意程式 伯格(-4)")) {
      baseAtk -= 4;
    }
    // Wave 7 Neutral Spells Buffs/Debuffs
    if (unit.equipments.includes("天黑黑")) {
      baseAtk = baseAtk - 1;
    }
    if (unit.equipments.includes("一根香蕉")) {
      baseAtk = baseAtk - 1;
    }
    if (unit.equipments.includes("前線號角") && unit.attacking) {
      baseAtk += 1;
    }
    if (unit.equipments.some(e => e.includes("我要打5個"))) {
      if (zone) {
        const isPlayer = zone.startsWith("player_");
        const oppFrontZone = isPlayer ? "enemy_front" : "player_front";
        const x = field[oppFrontZone].filter(u => u !== null).length;
        baseAtk += x;
      }
    }
  }

  // 4. 血戰幫狼牙棒 (裝備)
  if (unit.equipments && unit.equipments.includes("血戰幫狼牙棒")) {
    let maceCount = 0;
    for (const z of ["player_front", "player_back", "enemy_front", "enemy_back"]) {
      field[z].forEach(u => {
        if (u && u.equipments && u.equipments.includes("血戰幫狼牙棒")) {
          maceCount++;
        }
      });
    }
    baseAtk += maceCount;
  }

  // 5. 爺爺 Buff 過期判斷
  if (unit.grandpaBuffExpiresTurn && turn >= unit.grandpaBuffExpiresTurn) {
    unit.atkModifier = Math.max(0, (unit.atkModifier || 0) - 2);
    delete unit.grandpaBuffExpiresTurn;
  }

  // 拿波崙 (R-ART-0011) 同列敵方單位減攻效果
  if (zone && lane !== undefined) {
    const oppZones = zone.startsWith("player_") ? ["enemy_front", "enemy_back"] : ["player_front", "player_back"];
    let hasNapoleon = false;
    for (const oz of oppZones) {
      if (field[oz] && field[oz][lane]) {
        const oppUnit = field[oz][lane];
        const oc = oppUnit.card;
        if (oc && (oc.id === "R-ART-0011" || oc.name?.includes("拿波崙"))) {
          hasNapoleon = true;
          break;
        }
      }
    }
    if (hasNapoleon) {
      baseAtk = baseAtk - 2;
    }
  }

  // 博物館 (ART-0019) 防守加成效果
  if (isBeingAttacked && zone) {
    const owner = zone.startsWith("player_") ? "player" : "enemy";
    const fieldEl = owner === "player" ? document.getElementById("playerField") : document.getElementById("enemyField");
    if (fieldEl && fieldEl.dataset.card) {
      try {
        const fieldCard = JSON.parse(fieldEl.dataset.card);
        const isMuseum = fieldCard.id === "ART-0019" || fieldCard.name?.includes("博物館");
        const isArt = c && (c.faction === "藝術品" || c.id?.includes("ART") || c.race?.includes("藝術品"));
        const isShield = window.XLW_isShieldUnit(unit, zone, lane);
        if (isMuseum && isArt && isShield) {
          baseAtk += 1;
        }
      } catch (e) {}
    }
  }

  // 迴旋飛斧 (最終攻擊力兩倍計算)
  if (unit.equipments && (unit.equipments.includes("迴旋飛斧") || unit.equipments.some(e => e.includes("迴旋飛斧")))) {
    baseAtk = baseAtk * 2;
  }

  return baseAtk + rainbowBonus;
}

function isUnitMagicImmune(unit, zone, idx) {
  if (!unit || !unit.card) return false;
  let z = zone;
  let i = idx;
  if (!z || i === undefined || i === -1) {
    for (const testZone of ["player_front", "player_back", "enemy_front", "enemy_back"]) {
      const testIdx = field[testZone].indexOf(unit);
      if (testIdx >= 0) {
        z = testZone;
        i = testIdx;
        break;
      }
    }
  }
  if (z && i !== undefined && i !== -1 && window.isUnitSilenced(unit, z, i)) return false;

  if (z) {
    const side = z.startsWith("player_") ? "player" : "enemy";
    if (hasFriendlyHacker(side) && isRogueUnit(unit)) {
      return true;
    }
  }

  // 中星城菁英 (R-NMS-0075 / SSSR-NMS-0075) 正前方單位魔抗效果
  if (z === "player_front" && i !== undefined && i !== -1) {
    const backUnit = field.player_back[i];
    if (backUnit && (backUnit.card?.id === "R-NMS-0075" || backUnit.card?.id === "SSSR-NMS-0075" || backUnit.card?.name?.includes("中星城菁英")) && !window.isUnitSilenced(backUnit, "player_back", i)) {
      return true;
    }
  }
  if (z === "enemy_front" && i !== undefined && i !== -1) {
    const backUnit = field.enemy_back[i];
    if (backUnit && (backUnit.card?.id === "R-NMS-0075" || backUnit.card?.id === "SSSR-NMS-0075" || backUnit.card?.name?.includes("中星城菁英")) && !window.isUnitSilenced(backUnit, "enemy_back", i)) {
      return true;
    }
  }

  if (unit.magicImmune) return true;
  if (unit.magicImmuneUntilTurn && unit.magicImmuneUntilTurn >= turn) return true;
  
  const c = unit.card;
  const isArtMagicImmune = c.id === "ART-0005" || c.name?.includes("鎧甲擺件") ||
                           c.id === "ART-0009" || c.name?.includes("梵老爹") ||
                           c.id === "ART-0015" || c.name?.includes("番茄之子") ||
                           c.id === "ART-0016" || c.name?.includes("摩艾石巨像") ||
                           c.id === "R-ART-0001" || c.name?.includes("萌娜麗莎") ||
                           c.id === "SR-ART-0026" || c.name?.includes("和平使者 耶叔") ||
                           c.id === "R-ART-0012" || c.name?.includes("頸部痠痛的女人") ||
                           c.id === "R-ART-0038" || c.name?.includes("金衣服的女人") ||
                           c.id === "R-ART-0042" || c.name?.includes("哭泣女人") ||
                           c.id === "SR-ART-0037" || c.name?.includes("率性龐克") ||
                           c.id === "SR-ART-0059" || c.id === "ASSR-ART-0059" || c.name?.includes("始皇帝") ||
                           c.id === "C-ART-0060" || c.name?.includes("戰士陶俑") ||
                           c.id === "R-ART-0046" || c.name?.includes("牆壁上的塗鴉") ||
                           c.id === "R-VRT-0035" || c.id === "SSR-VRT-0035" || c.name?.includes("海王干貝") ||
                           c.id === "SSSR-NMS-0052" || c.name?.includes("潔白女神") || c.name?.includes("處女座");
  if (isArtMagicImmune) return true;
  
  return false;
}

async function destroyUnit(zone, idx, owner, shouldExile, isCombatDestruction = false) {
  const unit = field[zone][idx];
  if (!unit) return;

  // R-FMS-0022 萬字大叔 代替破壞效果
  if (unit.card && window.isMahjongUnit(unit.card) && !unit.card.name?.includes("萬字大叔")) {
    const isPlayer = zone.startsWith("player_");
    const sidePrefix = isPlayer ? "player_" : "enemy_";
    const hasWanzi = [sidePrefix + "front", sidePrefix + "back"].some(z => {
      return field[z].some((u, i) => u && u.card && (u.card.id === "R-FMS-0022" || u.card.name?.includes("萬字大叔")) && !window.isUnitSilenced(u, z, i));
    });
    const wanziUsed = isPlayer ? window.XLW_wanziUsedThisTurn : window.XLW_enemyWanziUsedThisTurn;
    if (hasWanzi && !wanziUsed) {
      const otherMahjongs = [];
      [sidePrefix + "front", sidePrefix + "back"].forEach(z => {
        field[z].forEach((u, i) => {
          if (u && u !== unit && u.card && window.isMahjongUnit(u.card)) {
            otherMahjongs.push({ zone: z, idx: i, name: u.card.name });
          }
        });
      });
      if (otherMahjongs.length > 0) {
        if (isPlayer) {
          const confirmSubstitute = await showXLWConfirm("萬字大叔 效果發動", `我方麻將單位【${unit.card.name}】即將被破壞，是否發動【萬字大叔】效果，選擇另一個我方麻將單位代替破壞？`);
          if (confirmSubstitute) {
            const choices = otherMahjongs.map((item, idx) => ({ text: `${item.name} (${item.zone.includes("front") ? "前排" : "後排"}${item.idx + 1})`, value: idx }));
            const chosen = await showXLWChoiceModal("選擇代替破壞的麻將", "請選擇代替破壞的單位：", choices);
            if (chosen !== null && chosen !== undefined) {
              const target = otherMahjongs[chosen];
              window.XLW_wanziUsedThisTurn = true;
              logBattle(`✨ 萬字大叔 效果：我方【${target.name}】代替【${unit.card.name}】被破壞！`);
              await destroyUnit(target.zone, target.idx, "player", false);
              return; // Skip original destruction!
            }
          }
        } else {
          // AI substitute
          const target = otherMahjongs[0];
          window.XLW_enemyWanziUsedThisTurn = true;
          logBattle(`✨ 對手 萬字大叔 效果：對手【${target.name}】代替【${unit.card.name}】被破壞！`);
          await destroyUnit(target.zone, target.idx, "enemy", false);
          return; // Skip original destruction!
        }
      }
    }
  }

  // R-FMS-0001 三寶 戰鬥被破壞效果
  if (isCombatDestruction && unit.card && (unit.card.id === "R-FMS-0001" || unit.card.name?.includes("三寶")) && !window.isUnitSilenced(unit, zone, idx)) {
    const isPlayer = zone.startsWith("player_");
    const oppPrefix = isPlayer ? "enemy_" : "player_";
    const oppOwner = isPlayer ? "enemy" : "player";
    let oppTargetZone = null, oppTargetIdx = -1;
    if (field[oppPrefix + "front"][idx]) {
      oppTargetZone = oppPrefix + "front";
      oppTargetIdx = idx;
    } else if (field[oppPrefix + "back"][idx]) {
      oppTargetZone = oppPrefix + "back";
      oppTargetIdx = idx;
    }
    if (oppTargetZone) {
      logBattle(`✨ 三寶 效果發動：前方敵方單位【${field[oppTargetZone][oppTargetIdx].card.name}】同歸於盡破壞！`);
      await destroyUnit(oppTargetZone, oppTargetIdx, oppOwner, false);
    }
  }

  // C-ORC-0060 作弊獸人 戰鬥失敗額外分數
  if (isCombatDestruction && unit.card && (unit.card.id === "C-ORC-0060" || unit.card.name?.includes("作弊獸人")) && !window.isUnitSilenced(unit, zone, idx)) {
    const isPlayer = zone.startsWith("player_");
    if (isPlayer) {
      playerBonusScore += 3;
      logBattle("✨ 作弊獸人 效果發動：戰鬥失敗，我方額外獎勵 +3★！");
    } else {
      enemyBonusScore += 3;
      logBattle("✨ 對手 作弊獸人 效果發動：戰鬥失敗，對手額外獎勵 +3★！");
    }
    renderScore();
  }

  // 金幣小魔 (BRV-0002) 戰鬥被破壞給敵方 +2 獎勵
  if (isCombatDestruction && unit.card && (unit.card.id === "BRV-0002" || unit.card.id === "R-BRV-0002" || unit.card.id === "SSSR-BRV-0002" || unit.card.name?.includes("金幣小魔")) && !window.isUnitSilenced(unit, zone, idx)) {
    const isPlayer = zone.startsWith("player_");
    if (isPlayer) {
      enemyBonusScore += 2;
      logBattle("✨ 金幣小魔 效果發動：我方金幣小魔被戰鬥破壞，對手額外獎勵 +2★！");
    } else {
      playerBonusScore += 2;
      logBattle("✨ 對手 金幣小魔 效果發動：對手金幣小魔被戰鬥破壞，我方額外獎勵 +2★！");
    }
    renderScore();
  }

  // 捲髮魔 (R-BRV-0008) 防守戰鬥被破壞：回收 1★ 以下魔法
  if (isCombatDestruction && unit.card && (unit.card.id === "R-BRV-0008" || unit.card.id === "BRV-0008" || unit.card.name?.includes("捲髮魔")) && !window.isUnitSilenced(unit, zone, idx)) {
    const isPlayer = zone.startsWith("player_");
    const isDefending = (isPlayer && !isMyTurn) || (!isPlayer && isMyTurn);
    if (isDefending) {
      await window.xlwRetrieveMagicFromGrave(isPlayer, 1);
    }
  }

  // R-ORC-0023 阿姨獸人 敵方小旅人被擊破效果
  if (isCombatDestruction && unit.card && (unit.card.id === "TOKEN_TRAVELER" || unit.card.name?.includes("小旅人"))) {
    const isPlayerTraveler = zone.startsWith("player_");
    const checkSidePrefix = isPlayerTraveler ? "enemy_" : "player_";
    const hasAuntOrc = [checkSidePrefix + "front", checkSidePrefix + "back"].some(z => {
      return field[z].some((u, i) => u && u.card && (u.card.id === "R-ORC-0023" || u.card.name?.includes("阿姨獸人")) && !window.isUnitSilenced(u, z, i));
    });
    if (hasAuntOrc) {
      if (isPlayerTraveler) {
        enemyBonusScore += 1;
        logBattle("✨ 對手 阿姨獸人 效果發動：我方小旅人被戰鬥破壞，對手額外獎勵 +1★！");
      } else {
        playerBonusScore += 1;
        logBattle("✨ 我方 阿姨獸人 效果發動：對手小旅人被戰鬥破壞，我方額外獎勵 +1★！");
      }
      renderScore();
    }
  }

  // 加命蘑菇 替代破壞效果
  const xlwMushroomIsPlayer = zone.startsWith("player_");
  const xlwMushroomSidePrefix = xlwMushroomIsPlayer ? "player_" : "enemy_";
  const xlwMushroomIsTributeFreeVirtual = unit.card && getCardTributeCost(unit.card) === 0 && window.isVirtualUnit(unit);
  if (xlwMushroomIsTributeFreeVirtual && unit.card.id !== "R-VRT-0053" && !unit.card.name?.includes("加命蘑菇")) {
    let lifeMushroomZone = null, lifeMushroomIdx = -1;
    for (const r of ["front", "back"]) {
      const z = xlwMushroomSidePrefix + r;
      for (let i = 0; i < 5; i++) {
        const u = field[z][i];
        if (u && u.card && (u.card.id === "R-VRT-0053" || u.card.name?.includes("加命蘑菇")) && !window.isUnitSilenced(u, z, i)) {
          lifeMushroomZone = z;
          lifeMushroomIdx = i;
          break;
        }
      }
      if (lifeMushroomZone) break;
    }
    if (lifeMushroomZone && lifeMushroomIdx !== -1) {
      logBattle(`✨ 加命蘑菇 效果發動：代替【${unit.card.name}】被破壞！`);
      await destroyUnit(lifeMushroomZone, lifeMushroomIdx, owner, false, false);
      return;
    }
  }

  // 實習畫家 替代破壞效果
  const isApprenticePainter = unit.card && (unit.card.id === "R-ART-0032" || unit.card.id === "ART-0032" || unit.card.name?.includes("實習畫家"));
  if (isApprenticePainter) {
    const isPlayer = zone.startsWith("player_");
    if (isPlayer) {
      const pigmentInHand = hand.filter(c => c && isPigmentCard(c));
      if (pigmentInHand.length > 0) {
        const confirmSubstitute = await showXLWConfirm("實習畫家 效果發動", "實習畫家即將被破壞，是否棄置手牌中的 1 張顏料卡代替破壞？");
        if (confirmSubstitute) {
          const choices = pigmentInHand.map(c => ({ text: `${c.name} (${c.id})`, value: hand.indexOf(c) }));
          const discardIdx = await showXLWChoiceModal("選擇顏料卡棄置", "請點選要送入墓地的手牌顏料卡：", choices);
          if (discardIdx !== null && discardIdx !== undefined) {
            const discarded = hand[discardIdx];
            hand.splice(discardIdx, 1);
            graveyard.push(discarded);
            logBattle(`✨ 實習畫家 效果發動：棄置手牌中的 ${discarded.name}，代替破壞！`);
            if (isMultiplayer) {
              sendFullGameStateToOpponent();
            }
            render();
            return;
          }
        }
      }
    } else {
      const enemyHand = window.XLW_ENEMY?.hand || [];
      const pigmentInEnemyHand = enemyHand.filter(c => c && isPigmentCard(c));
      if (pigmentInEnemyHand.length > 0) {
        const discarded = pigmentInEnemyHand[Math.floor(Math.random() * pigmentInEnemyHand.length)];
        enemyHand.splice(enemyHand.indexOf(discarded), 1);
        window.XLW_ENEMY.grave.push(discarded);
        logBattle(`✨ 對手 實習畫家 效果發動：對手棄置其手牌中的 ${discarded.name}，代替破壞！`);
        render();
        return;
      }
    }
  }

  if (isUnitIndestructible(unit, zone, idx, isCombatDestruction)) {
    logBattle(`✨ ${unitName(unit)} 具有破壞抗性，免於被破壞！`);
    return;
  }

  // 1. 尋找格子並播放 CSS 碎裂與紅光粒子動畫
  const slot = document.querySelector(`[data-zone="${zone}"][data-index="${idx}"]`);
  if (slot) {
    const cardEl = slot.querySelector(".card");
    if (cardEl) {
      cardEl.classList.add("xlw-card-shatter");
      
      const sparks = document.createElement("div");
      sparks.className = "shatter-sparks";
      slot.appendChild(sparks);

      // 1. 如月車站 生死倒數保護
      if (unit.card.id === "R-VLG-0018" || unit.card.name?.includes("如月車站")) {
        if (countdownActive) {
          logBattle("✨ 如月車站 效果：生死倒數階段此卡不會被破壞！");
          if (sparks) sparks.remove();
          return;
        }
      }

      // 2. 逃脫 效果阻斷送墓
      if (isCombatDestruction) {
        if (owner === "player" && window.XLW_playerCatEscapeTurn === turn) {
          hand.push(unit.card);
          logBattle(`✨ 逃脫 效果：被破壞的單位【${unit.card.name}】回到手牌！`);
          field[zone][idx] = null;
          if (sparks) sparks.remove();
          render();
          return;
        }
        if (owner === "enemy" && window.XLW_enemyCatEscapeTurn === turn) {
          if (!window.XLW_ENEMY) window.XLW_ENEMY = { hand: [] };
          if (!window.XLW_ENEMY.hand) window.XLW_ENEMY.hand = [];
          window.XLW_ENEMY.hand.push(unit.card);
          logBattle(`✨ 對手 逃脫 效果：被破壞的單位【${unit.card.name}】回到對手手牌！`);
          field[zone][idx] = null;
          if (sparks) sparks.remove();
          render();
          return;
        }
      }

      // 3. 水鬼 效果觸發 (戰鬥與效果破壞之同歸於盡)
      if (unit.card.id === "R-VLG-0038" || unit.card.name?.includes("水鬼")) {
        const isPlayer = zone.startsWith("player_");
        const oppOwner = isPlayer ? "enemy" : "player";
        const oppZones = isPlayer ? ["enemy_front", "enemy_back"] : ["player_front", "player_back"];
        
        if (isCombatDestruction) {
          let oppUnit = null;
          let oppZone = null;
          for (const z of oppZones) {
            if (field[z][idx]) {
              oppUnit = field[z][idx];
              oppZone = z;
              break;
            }
          }
          if (oppUnit) {
            logBattle(`✨ 水鬼 效果發動：與敵方單位【${oppUnit.card.name}】同歸於盡！`);
            await destroyUnit(oppZone, idx, oppOwner, false, false);
          }
        } else if (window.XLW_currentEffectSourceUnit) {
          const src = window.XLW_currentEffectSourceUnit;
          let srcZone = null;
          let srcIdx = -1;
          for (const z of oppZones) {
            const i = field[z].indexOf(src);
            if (i >= 0) {
              srcZone = z;
              srcIdx = i;
              break;
            }
          }
          if (srcZone && srcIdx >= 0) {
            logBattle(`✨ 水鬼 效果發動：使破壞牠的敵方單位【${src.card.name}】被破壞！`);
            await destroyUnit(srcZone, srcIdx, oppOwner, false, false);
          }
        }
      }

    }
  }

  // 2. 等待動畫完整播放結束 (0.8 秒)
  await sleep(800);

  // 3. 清理火花層
  if (slot) {
    const sparks = slot.querySelector(".shatter-sparks");
    if (sparks) sparks.remove();
  }

  // 4. 送墓或除外並真正清空格子
  if (unit.equipments && unit.equipments.length > 0) {
    unit.equipments.forEach(eqName => {
      if ((eqName === "賞金" || eqName.includes("賞金")) && isCombatDestruction) {
        if (zone.startsWith("player_")) {
          enemyBonusScore += 3;
          logBattle(`💥 附有【賞金】的我方單位 ${unit.card.name} 被戰鬥破壞：對手額外獲得 +3★ 獎勵！`);
        } else {
          playerBonusScore += 3;
          logBattle(`💥 附有【賞金】的敵方單位 ${unit.card.name} 被戰鬥破壞：我方額外獲得 +3★ 獎勵！`);
        }
        renderScore();
      }
      const isTempMagic = ["削弱藥水", "振奮藥水", "塗毒", "睡眠反擊拳"].some(x => eqName.includes(x));
      if (isTempMagic) return;
      const cardObj = allCards.find(c => c && c.name === eqName) || { name: eqName, type: "magic" };
      const cloned = structuredClone(cardObj);
      cloned.type = "magic";
      if (owner === "player") {
        graveyard.push(cloned);
        logBattle(`✨ 裝備卡 ${cloned.name} 隨著單位被破壞進入我方墓地。`);
      } else {
        window.XLW_ENEMY.grave.push(cloned);
        enemyGraveyard = window.XLW_ENEMY.grave;
        logBattle(`✨ 裝備卡 ${cloned.name} 隨著單位被破壞進入對手墓地。`);
      }
    });
    unit.equipments = [];
  }

  if (unit.card.name === "大耳賊" || unit.card.id === "NEU-0064") {
    if (zone.startsWith("player_")) {
      enemyBonusScore += 2;
      logBattle(`💥 大耳賊 被戰鬥破壞：對手額外獎勵 +2 ★！`);
    } else {
      playerBonusScore += 2;
      logBattle(`💥 大耳賊 被戰鬥破壞：我方額外獎勵 +2 ★！`);
    }
  }

  // 牆壁上的香蕉 (SR-ART-0025 / SSR-ART-0025) 被破壞得分
  if (unit.card.id === "SR-ART-0025" || unit.card.id === "SSR-ART-0025" || unit.card.name?.includes("牆壁上的香蕉")) {
    if (zone.startsWith("player_")) {
      playerBonusScore += 3;
      logBattle(`💥 我方 牆壁上的香蕉 被破壞：我方額外獎勵 +3 ★！`);
    } else {
      enemyBonusScore += 3;
      logBattle(`💥 對手 牆壁上的香蕉 被破壞：對手額外獎勵 +3 ★！`);
    }
  }

  // 無形暗影 雙子座 戰鬥破壞效果
  let geminiReturnedToHand = false;
  const isShadowGemini = unit.card && (unit.card.id === "SSSR-NMS-0061" || unit.card.name?.includes("無形暗影") || unit.card.name?.includes("雙子座 莉子") || unit.card.name?.includes("雙子座 妮子"));
  if (isShadowGemini && isCombatDestruction) {
    if (owner === "player") {
      const geminiCard = graveyard.find(c => c && (c.id === "SSSR-NMS-0059" || c.id === "SSSR-NMS-0060" || c.name?.includes("亞子") || c.name?.includes("平子") || c.name?.includes("雙面幻影")));
      let emptySlot = null;
      for (const z of ["player_front", "player_back"]) {
        const idx = field[z].findIndex(u => !u);
        if (idx >= 0) {
          emptySlot = { zone: z, idx: idx };
          break;
        }
      }
      if (geminiCard && emptySlot) {
        const triggerSummon = await showXLWConfirm("【無形暗影 雙子座】效果發動", `是否發動效果，特殊召喚墓地中的【${geminiCard.name}】，並將【${unit.card.name}】回收至手牌？`);
        if (triggerSummon) {
          graveyard.splice(graveyard.indexOf(geminiCard), 1);
          field[emptySlot.zone][emptySlot.idx] = {
            card: geminiCard,
            tapped: false,
            attacking: false,
            target: null,
            summonedTurn: turn,
            summonedZone: emptySlot.zone
          };
          hand.push(unit.card);
          geminiReturnedToHand = true;
          logBattle(`✨ 無形暗影 雙子座 效果：從墓地特殊召喚【${geminiCard.name}】，並將【${unit.card.name}】加入手牌！`);
          if (isMultiplayer && ws) {
            ws.send(JSON.stringify({
              action: "shadow_gemini_sync",
              geminiCard: geminiCard,
              zone: emptySlot.zone,
              idx: emptySlot.idx,
              shadowCard: unit.card
            }));
          }
          render();
        }
      }
    } else {
      const oppGrave = window.XLW_ENEMY.grave || [];
      const geminiCard = oppGrave.find(c => c && (c.id === "SSSR-NMS-0059" || c.id === "SSSR-NMS-0060" || c.name?.includes("亞子") || c.name?.includes("平子") || c.name?.includes("雙面幻影")));
      let emptySlot = null;
      for (const z of ["enemy_front", "enemy_back"]) {
        const idx = field[z].findIndex(u => !u);
        if (idx >= 0) {
          emptySlot = { zone: z, idx: idx };
          break;
        }
      }
      if (geminiCard && emptySlot && !isMultiplayer) {
        oppGrave.splice(oppGrave.indexOf(geminiCard), 1);
        field[emptySlot.zone][emptySlot.idx] = {
          card: geminiCard,
          tapped: false,
          attacking: false,
          target: null,
          summonedTurn: turn,
          summonedZone: emptySlot.zone
        };
        window.XLW_ENEMY.hand.push(unit.card);
        geminiReturnedToHand = true;
        logBattle(`✨ 對手 無形暗影 雙子座 效果：從墓地特殊召喚【${geminiCard.name}】，並將【${unit.card.name}】加入手牌！`);
        render();
      }
    }
  }

  if (unit.card.name.includes("小旅人") || unit.card.name.includes("尋寶小旅人") || unit.card.id === "TOKEN_TRAVELER" || unit.card.id === "NEU-0010") {
    logBattle(`✨ ${unit.card.name} 被破壞，直接返回森林（消失在遊戲中）。`);
    
    let playerHasLandlord = ["player_front", "player_back"].some(z => field[z].some(u => u && (u.card?.id === "NEU-0036" || u.card?.name?.includes("恐怖的房東阿伯"))));
    let enemyHasLandlord = ["enemy_front", "enemy_back"].some(z => field[z].some(u => u && (u.card?.id === "NEU-0036" || u.card?.name?.includes("恐怖的房東阿伯"))));
    if (playerHasLandlord) {
      playerBonusScore += 3;
      logBattle("✨ 我方的 恐怖的房東阿伯 效果觸發：偵測到小旅人被破壞，我方獲得 +3★ 房租獎勵！");
      renderScore();
    }
    if (enemyHasLandlord) {
      enemyBonusScore += 3;
      logBattle("✨ 對手的 恐怖的房東阿伯 效果觸發：偵測到小旅人被破壞，對手獲得 +3★ 房租獎勵！");
      renderScore();
    }
  } else if (geminiReturnedToHand) {
    // Already returned to hand by shadow Gemini trigger
  } else if (shouldExile) {
    logBattle(`✨ 被擊破的 ${unit.card.name} 被除外（不進入墓地）！`);
    exileCard(unit.card, owner);
  } else {
    // 如月車站 被破壞時釋放/破壞封存單位

          let confirmUse = true;
          if (isPlayerOwner) {
            confirmUse = await showXLWConfirm("亡靈醫院院長 效果", "是否發動【亡靈醫院院長】效果，進行還魂 3 並召喚我方墓地中的一個亡靈單位？");
          }
          if (confirmUse) {
            const success = await window.xlwExecuteSoulReturn(3, isPlayerOwner);
            if (success) {
              await window.xlwSummonUndeadFromGraveyard(isPlayerOwner);
            }
          }
        }
      }
    }
  }

  // 金幣蘑菇 (R-VRT-0055) 死亡觸發
  if (unit.card.id === "R-VRT-0055" || unit.card.name?.includes("金幣蘑菇")) {
    const isPlayerOwner = owner === "player";
    const sidePrefix = isPlayerOwner ? "player_" : "enemy_";
    let hasOtherMushroom = false;
    for (const r of ["front", "back"]) {
      const z = sidePrefix + r;
      for (let i = 0; i < 5; i++) {
        const u = field[z][i];
        if (u && u !== unit && u.card && (u.card.name?.includes("蘑菇") || u.card.name?.includes("魔菇"))) {
          hasOtherMushroom = true;
          break;
        }
      }
      if (hasOtherMushroom) break;
    }
    if (hasOtherMushroom) {
      if (isPlayerOwner) {
        playerBonusScore += 2;
        logBattle("✨ 金幣蘑菇 效果：金幣蘑菇被破壞，我方場上有其他蘑菇，獲得 +2★ 獎勵！");
      } else {
        enemyBonusScore += 2;
        logBattle("✨ 對手 金幣蘑菇 效果：金幣蘑菇被破壞，對手場上有其他蘑菇，獲得 +2★ 獎勵！");
      }
    } else {
      if (isPlayerOwner) {
        playerBonusScore += 1;
        logBattle("✨ 金幣蘑菇 效果：金幣蘑菇被破壞，我方獲得 +1★ 獎勵！");
      } else {
        enemyBonusScore += 1;
        logBattle("✨ 對手 金幣蘑菇 效果：金幣蘑菇被破壞，對手獲得 +1★ 獎勵！");
      }
    }
    renderScore();
  }

  // 千面病毒 (SR-VRT-0032 / SSSR-VRT-0032) 宿主死亡轉移
  if (unit.equipments && unit.equipments.includes("千面病毒") && unit.virusCard) {
    const vCard = unit.virusCard;
    const vOwner = unit.virusOwner || owner;
    const isPlayerVirus = vOwner === "player";
    const sidePrefix = isPlayerVirus ? "player_" : "enemy_";
    const hosts = [];
    for (const r of ["front", "back"]) {
      const z = sidePrefix + r;
      field[z].forEach((u, i) => {
        if (u && u !== unit && u.card && window.isVirtualUnit(u)) {
          hosts.push({ zone: z, idx: i, unit: u });
        }
      });
    }
    if (hosts.length > 0) {
      let selectedHost = null;
      if (isPlayerVirus) {
        const choices = hosts.map((item, i) => ({
          text: `${item.zone.includes("front") ? "前排" : "後排"}${item.idx + 1} 的 ${item.unit.card.name}`,
          value: i
        }));
        const chosenIdx = await showXLWChoiceModal("千面病毒 轉移效果", "請選擇轉移寄生合體的虛擬單位：", choices);
        if (chosenIdx !== null && chosenIdx !== undefined) {
          selectedHost = hosts[chosenIdx];
        }
      } else {
        selectedHost = hosts[Math.floor(Math.random() * hosts.length)];
      }
      if (selectedHost) {
        logBattle(`✨ 千面病毒 轉移：宿主【${unit.card.name}】被破壞，轉移寄生至【${selectedHost.unit.card.name}】！`);
        selectedHost.unit.equipments = selectedHost.unit.equipments || [];
        selectedHost.unit.equipments.push("千面病毒");
        selectedHost.unit.virusCard = vCard;
        selectedHost.unit.virusOwner = vOwner;
        render();
      }
    } else {
      logBattle("千面病毒 轉移：無其他虛擬單位可寄生，千面病毒消失。");
    }
  }

  await cleanUnitLeavingField(unit, zone);
  field[zone][idx] = null;
}

async function performSummonToSlot(zone, idx) {
  let isSpecialSummon = false;
  let specialCard = null;
  if (typeof zone === "object") {
    isSpecialSummon = true;
    specialCard = zone;
    zone = arguments[1];
    idx = arguments[2];
  }

  let summonMessageSent = false;
  let immediateEffectNegated = false;
  logDebug(`【點選放置】目標格子: ${zone}:${idx + 1}`);
  // 獻祭召喚與位置放置之強大對象參考匹配與 Fallback 恢復機制
  if (tributeWaitingPosition && selectedTributeCard !== null) {
    const isGraveyardSummon = (selectedHandForSummon === "GRAVEYARD_UMBRELLA" || 
                               selectedHandForSummon === "GRAVEYARD_WELL" || 
                               selectedHandForSummon === "GRAVEYARD_GHOST_MEOW");
    let isDifferentHandCard = false;
    if (typeof selectedHandForSummon === "number" || (typeof selectedHandForSummon === "string" && !isNaN(selectedHandForSummon))) {
      const currentCard = hand[Number(selectedHandForSummon)];
      if (currentCard && currentCard.id !== selectedTributeCard.id) {
        isDifferentHandCard = true;
      }
    }
    
    if (!isGraveyardSummon && !isDifferentHandCard) {
      const idxInHand = hand.indexOf(selectedTributeCard);
      if (idxInHand >= 0) {
        selectedHandForSummon = idxInHand;
        logDebug(`【對象匹配】實體匹配成功！手牌 Index: ${selectedHandForSummon}`);
      } else {
        const matchIdx = hand.findIndex(c => c && c.id === selectedTributeCard.id);
        if (matchIdx >= 0) {
          selectedHandForSummon = matchIdx;
          logDebug(`【對象匹配】ID 匹配成功！手牌 Index: ${selectedHandForSummon}`);
        } else {
          const foundIdx = hand.findIndex(c => c && getCardTributeCost(c) > 0);
          if (foundIdx >= 0) {
            selectedHandForSummon = foundIdx;
            logDebug(`【對象匹配】兜底 Fallback 匹配成功！手牌 Index: ${selectedHandForSummon}`);
          } else {
            logDebug("【放置終止】祭品已送墓，但在手牌中找不到任何獻祭卡牌！");
            return;
          }
        }
      }
    }
  }

  if (selectedHandForSummon === null) {
    logDebug("【放置終止】當前 selectedHandForSummon 為 null！");
    return;
  }

  if (isMultiplayer && !isMyTurn && !window.XLW_bypassTurnCheck) {
    setStatus("對手回合中，請稍候！");
    logDebug("【放置終止】非我方回合！");
    return;
  }

  const canSum = window.XLW_bypassNormalSummonLimit ||
                 (phase === "召喚階段" && !normalSummonUsed) || 
                 (phase === "戰術佈陣" && !tacticalSummonUsed);
  if (!canSum) {
    setStatus("本回合已進行過單位召喚。");
    logDebug(`【放置終止】本回合召喚次數已用盡！(normalSummonUsed: ${normalSummonUsed})`);
    clearModes();
    render();
    return;
  }

  let card = null;
  try {

  if (selectedHandForSummon === "TRAVELER") {
    // 召喚小旅人
    field[zone][idx] = {
      card: structuredClone(LITTLE_TRAVELER),
      tapped: false,
      attacking: false,
      target: null,
      summonedTurn: turn,
      summonedZone: zone
    };
    card = field[zone][idx].card;
    if (phase === "召喚階段" && !window.XLW_bypassTurnCheck) normalSummonUsed = true;
    if (phase === "戰術佈陣" && !window.XLW_bypassTurnCheck) tacticalSummonUsed = true;
    logBattle(`從森林召喚小旅人到 ${zone.includes("front") ? "前排" : "後排"}${idx + 1}`);
    playTravelerSummonAnimation("#playerForest", zone, idx);
    if (zone.startsWith("player_")) {
      window.XLW_playerSummonCountThisTurn = (window.XLW_playerSummonCountThisTurn || 0) + 1;
      window.XLW_playerActionsPerformedThisTurn = true;
    } else {
      window.XLW_enemySummonCountThisTurn = (window.XLW_enemySummonCountThisTurn || 0) + 1;
    }
  } else {
    // 召喚手牌
    const isTributeSummon = (tributeWaitingPosition && selectedTributeCard !== null);
    let handIndex = Number(selectedHandForSummon);
    
    card = null;
    if (selectedHandForSummon === "GRAVEYARD_UMBRELLA" || selectedHandForSummon === "GRAVEYARD_WELL" || selectedHandForSummon === "GRAVEYARD_GHOST_MEOW") {
      card = window.XLW_tempGraveCard;
    } else {
      if (handIndex < 0 || handIndex >= hand.length || !hand[handIndex]) {
        if (tributeWaitingPosition) {
          const foundIdx = hand.findIndex(c => c && getCardTributeCost(c) > 0);
          if (foundIdx >= 0) {
            handIndex = foundIdx;
          }
        }
      }
      if (handIndex >= 0 && hand[handIndex]) {
        card = hand[handIndex];
      }
    }

    if (!card) {
      setStatus("找不到手牌中的卡牌，請點選重新召喚。");
      return;
    }

    const isLyingOrc = card.id === "R-ORC-0031" || card.name?.includes("躺平獸人");
    field[zone][idx] = {
      card: structuredClone(card),
      tapped: isLyingOrc,
      attacking: false,
      target: null,
      summonedTurn: turn,
      summonedZone: zone
    };
    field[zone][idx].card.creator = "player";
    if (zone.startsWith("player_")) {
      window.XLW_playerSummonCountThisTurn = (window.XLW_playerSummonCountThisTurn || 0) + 1;
    } else {
      window.XLW_enemySummonCountThisTurn = (window.XLW_enemySummonCountThisTurn || 0) + 1;
    }

    const isExtraSum = (selectedHandForSummon === "GRAVEYARD_UMBRELLA" || 
                        selectedHandForSummon === "GRAVEYARD_WELL" || 
                        selectedHandForSummon === "GRAVEYARD_GHOST_MEOW" ||
                        window.XLW_bypassNormalSummonLimit);
    if (isExtraSum && zone.startsWith("player_")) {
      window.XLW_turnExtraSummonsCount = (window.XLW_turnExtraSummonsCount || 0) + 1;
      logDebug(`【額外召喚追蹤】我方進行了額外召喚！本回合已額外召喚 ${window.XLW_turnExtraSummonsCount} 個單位。`);
    }

    if (selectedHandForSummon === "GRAVEYARD_UMBRELLA" || selectedHandForSummon === "GRAVEYARD_WELL" || selectedHandForSummon === "GRAVEYARD_GHOST_MEOW") {
      window.XLW_tempGraveCard = null;
      window.XLW_bypassNormalSummonLimit = false;
      logBattle(`特殊召喚墓地單位 ${card.name} 到 ${zone.includes("front") ? "前排" : "後排"}${idx + 1}`);
    } else {
      hand.splice(handIndex, 1);
      if (window.XLW_bypassNormalSummonLimit) {
        window.XLW_bypassNormalSummonLimit = false;
        logDebug(`【連動額外召喚】成功！不佔用本回合常規召喚額度。`);
        
        const isPlayer = zone.startsWith("player_");
        const checkZonePrefix = isPlayer ? "player_" : "enemy_";
        const prOnField = field[checkZonePrefix + "front"].concat(field[checkZonePrefix + "back"]).some(u => {
          return u && u.card && (u.card.id === "CAT_005" || u.card.id === "R-CAT-0041" || u.card.name.includes("公關姐姐喵"));
        });
        if (prOnField) {
          if (isPlayer) {
            playerBonusScore += 1;
            logBattle(`✨ 公關姐姐喵 效果：我方進行額外召喚單位，我方額外獎勵 +1 分！`);
          } else {
            enemyBonusScore += 1;
            logBattle(`✨ 公關姐姐喵 效果：對手進行額外召喚單位，對手額外獎勵 +1 分！`);
          }
          renderScore();
        }
      } else {
        if (phase === "召喚階段") normalSummonUsed = true;
        if (phase === "戰術佈陣") tacticalSummonUsed = true;
      }
      logDebug(`【召喚成功】卡牌: ${card.name} 已成功放置於 ${zone}:${idx + 1}！手牌扣減成功。`);
      logBattle(`召喚單位 ${card.name} 到 ${zone.includes("front") ? "前排" : "後排"}${idx + 1}`);
    }
    
    // 黑心網路賣家 (R-ART-0036) 我方召喚藝術品單位到敵方場上時，額外得分 +1★
    if (zone.startsWith("enemy_") && hasBlackMerchantOnField("player")) {
      const isArtwork = card && (card.faction === "藝術品" || card.deck === "藝術品" || card.id?.includes("ART"));
      if (isArtwork) {
        playerBonusScore += 1;
        logBattle("✨ 黑心網路賣家 效果：我方將藝術品單位召喚至敵方場上，我方得分 +1 ★！");
      }
    }
    
    // 博物館館長 迪斯麥 (SR-ART-0039) 主要階段限一次，打出/召喚非人型藝術品時可將館長橫置
    if (zone.startsWith("player_") && isNonHumanoidArtworkCard(card)) {
      let directorUnit = null;
      for (const z of ["player_front", "player_back"]) {
        field[z].forEach(u => {
          if (u && (u.card?.id === "SR-ART-0039" || u.card?.name?.includes("博物館館長")) && !u.tapped && u.directorEffectUsedTurn !== turn) {
            directorUnit = u;
          }
        });
      }
      if (directorUnit) {
        const confirmTap = await showXLWConfirm("博物館館長 迪斯麥 效果", `我方召喚了非人型藝術品，是否發動【博物館館長 迪斯麥】效果將其橫放？`);
        if (confirmTap) {
          directorUnit.tapped = true;
          directorUnit.directorEffectUsedTurn = turn;
          logBattle("✨ 博物館館長 迪斯麥 效果：橫置自身！");
          if (isMultiplayer) {
            sendFullGameStateToOpponent();
          }
          render();
        }
      }
    }
    
    immediateEffectNegated = false;
    summonMessageSent = false;

    if (isMultiplayer) {
      if (!window.XLW_isFoodBallSummoning) {
        summonMessageSent = true;
        const isFromHand = (selectedHandForSummon !== "TRAVELER" &&
                            selectedHandForSummon !== "GRAVEYARD_UMBRELLA" &&
                            selectedHandForSummon !== "GRAVEYARD_WELL" &&
                            selectedHandForSummon !== "GRAVEYARD_GHOST_MEOW");
        ws.send(JSON.stringify({
          action: "summon",
          card: field[zone][idx].card,
          zone: zone,
          idx: idx,
          fromHand: isFromHand,
          tributes: window.XLW_lastTributes || []
        }));
        window.XLW_lastTributes = null;

        const response = await new Promise(resolve => {
          window.XLW_pendingSummonCounterResolve = resolve;
          // Safety timeout: 12 seconds fallback
          setTimeout(() => {
            if (window.XLW_pendingSummonCounterResolve === resolve) {
              console.log("[WS-Safety] Summon counter resolve timeout fired");
              window.XLW_pendingSummonCounterResolve = null;
              resolve({ countered: false });
            }
          }, 12000);
        });

        if (response.countered === true || response.countered === "true") {
          immediateEffectNegated = true;
          logBattle(`對手發動了手牌中的【星空】！召喚的【${card.name}】被破壞，效果無效。`);
          await destroyUnit(zone, idx, "player", false);
          render();
        } else if (response.countered === "interrupt") {
          immediateEffectNegated = true;
          logBattle(`對手發動了手牌中的【打斷你一下】！無效化了【${card.name}】的立即效果（單位保留）。`);
          render();
        }
      }
    } else {
      const oppHand = window.XLW_ENEMY?.hand || [];
      const hasStarry = oppHand.some(c => c && (c.id === "ART-0020" || c.name?.includes("星空")));
      const hasInterrupt = oppHand.some(c => c && (c.id === "R-NMG-0022" || c.id === "SSSR-NMG-0022" || c.name?.includes("打斷你一下")));
      const hasImm = hasImmediateEffect(card);

      if (hasStarry && Math.random() < 0.4) {
        const sIdx = oppHand.findIndex(c => c && (c.id === "ART-0020" || c.name?.includes("星空")));
        const sCard = oppHand.splice(sIdx, 1)[0];
        window.XLW_ENEMY.grave.push(sCard);
        enemyGraveyard = window.XLW_ENEMY.grave;

        immediateEffectNegated = true;
        logBattle(`✨ 對手 AI 發動了手牌中的【星空】！`);
        await destroyUnit(zone, idx, "player", false);
        render();
      } else if (hasInterrupt && hasImm && Math.random() < 0.5) {
        const sIdx = oppHand.findIndex(c => c && (c.id === "R-NMG-0022" || c.id === "SSSR-NMG-0022" || c.name?.includes("打斷你一下")));
        const sCard = oppHand.splice(sIdx, 1)[0];
        window.XLW_ENEMY.grave.push(sCard);
        enemyGraveyard = window.XLW_ENEMY.grave;

        immediateEffectNegated = true;
        logBattle(`✨ 對手 AI 發動了手牌中的【打斷你一下】！無效化了【${card.name}】的立即效果。`);
        render();
      }
    }

    if (!immediateEffectNegated) {
      let mushroomZone = null, mushroomIdx = -1;
      for (const z of ["enemy_front", "enemy_back"]) {
        field[z].forEach((u, i) => {
          if (u && u.card && (u.card.id === "SSSR-NMS-0042" || u.card.id === "R-NMS-0042" || u.card.name?.includes("驚嚇蘑菇"))) {
            mushroomZone = z;
            mushroomIdx = i;
          }
        });
      }
      if (mushroomZone && mushroomIdx !== -1 && hasImmediateEffect(card)) {
        logBattle(`✨ 對手場上的【驚嚇蘑菇】引爆！自毀並無效化了我方【${card.name}】的立即效果。`);
        await destroyUnit(mushroomZone, mushroomIdx, "enemy", false);
        immediateEffectNegated = true;
        render();
      }
    }

    if (!immediateEffectNegated && window.xlwIsImmediateEffectBlocked(card, true)) {
      immediateEffectNegated = true;
    }

    if (!immediateEffectNegated && card) {
      // SSSR-NMS-0031 小女魔導士庫路路
      if (card.id === "SSSR-NMS-0031" || card.name?.includes("庫路路") || card.name?.includes("魔導士")) {
        await window.xlwTriggerKururuEffect(zone.startsWith("player_"));
      }
      // 賢者 (BRV-0015) 召喚成功效果
      if (card.id === "BRV-0015" || card.id === "R-BRV-0015" || card.id === "SSSR-BRV-0015" || card.name?.includes("賢者")) {
        const isPlayer = zone.startsWith("player_");
        const sageUnit = field[zone][idx];
        await window.xlwTriggerSageEffect(isPlayer, sageUnit);
      }
      // SSSR-NMS-0053 幻想巨人 雙魚座
      if (card.id === "SSSR-NMS-0053" || card.name?.includes("雙魚座")) {
        if (zone.startsWith("player_")) {
          window.XLW_playerPiscesActive = true;
          logBattle("✨ 幻想巨人 雙魚座 效果發動：本次對決我方的所有戰術階段將不再被跳過！");
        } else {
          window.XLW_enemyPiscesActive = true;
          logBattle("✨ 對手 幻想巨人 雙魚座 效果發動：本次對決對手的所有戰術階段將不再被跳過！");
        }
      }
      // R-FMS-0007 大腸包小腸
      if (card.id === "R-FMS-0007" || card.name?.includes("大腸包小腸")) {
        const isPlayer = zone.startsWith("player_");
        const rev = await window.xlwRevealTopCards(1, isPlayer);
        if (rev.length > 0 && rev[0].type === "unit") {
          const revAtk = rev[0].attack ?? rev[0].atk ?? rev[0].power ?? 0;
          const oppSide = isPlayer ? "enemy" : "player";
          const oppZones = isPlayer ? ["enemy_front", "enemy_back"] : ["player_front", "player_back"];
          const targets = [];
          oppZones.forEach(z => {
            field[z].forEach((u, i) => {
              if (u && !isUnitEffectImmune(u, z, i)) {
                const uAtk = getUnitAtk(u, z, i);
                if (uAtk <= revAtk) {
                  targets.push({ zone: z, idx: i, unit: u });
                }
              }
            });
          });
          if (targets.length > 0) {
            if (isPlayer) {
              const choices = targets.map((t, idx) => ({ text: `${t.zone.includes("front") ? "前排" : "後排"}${t.idx + 1} 的 ${t.unit.card.name}`, value: idx }));
              const chosen = await showXLWChoiceModal("大腸包小腸 綁架效果", "選擇一個要綁架的敵方單位：", choices);
              if (chosen !== null && chosen !== undefined) {
                const t = targets[chosen];
                await kidnapCard(t.zone, t.idx, zone, idx);
                render();
              }
            } else {
              const t = targets[Math.floor(Math.random() * targets.length)];
              await kidnapCard(t.zone, t.idx, zone, idx);
              render();
            }
          }
        }
      }

      // R-FMS-0009 / R-FMS-0010 / R-FMS-0011 紅中哥/白板哥/發財哥
      if (card.name?.includes("紅中哥") || card.name?.includes("白板哥") || card.name?.includes("發財哥")) {
        const isPlayer = zone.startsWith("player_");
        const myDeck = isPlayer ? deck : (window.XLW_ENEMY.deck || []);
        if (myDeck.length > 0) {
          const rev = await window.xlwRevealTopCards(1, isPlayer);
          if (rev.length > 0) {
            const rName = rev[0].name || "";
            const isBrother = rName.includes("紅中哥") || rName.includes("白板哥") || rName.includes("發財哥");
            if (isBrother && rName !== card.name) {
              const emptySlots = isPlayer ? window.xlwGetEmptyPlayerSlots() : window.xlwGetEmptyEnemySlots();
              if (emptySlots.length > 0) {
                myDeck.pop(); // remove from top
                await window.xlwSpecialSummonUnit(rev[0], isPlayer);
                logBattle(`✨ 麻將兄弟連動：成功特殊召喚了【${rev[0].name}】！`);
                render();
              }
            }
          }
        }
      }

      // R-FMS-0013 北小妹

  flyEl.style.transform = "perspective(800px) translateZ(0px) rotate(360deg) scale(1)";

  setTimeout(() => {
    // 降落時觸發卡片格子綠色閃光特效
    targetSlot.classList.add("xlw-summon-flash-green");

    // 移除動態元素，釋放鎖定並重置渲染
    flyEl.remove();
    window.XLW_animatingSlots.delete(key);
    render();

    // 動態移除綠色閃光類別
    setTimeout(() => {
      targetSlot.classList.remove("xlw-summon-flash-green");
    }, 450);
  }, 850);
}

async function resolveUnitCombat(attZone, attIdx, defZone, defIdx, defenderOwner) {
  const attacker = field[attZone]?.[attIdx];
  const defender = field[defZone]?.[defIdx];

  if (!attacker || !defender) return "戰鬥目標不存在";

  let atkPower = getUnitAtk(attacker, attZone, attIdx);
  let defPower = getUnitAtk(defender, defZone, defIdx, true);

  // 黃金鬥士 金牛座 (SSSR-NMS-0040) 效果：與阻擋單位戰鬥時，阻擋單位的攻擊力視為其星數
  const attackerIsTaurus40 = attacker && attacker.card && (attacker.card.id === "SSSR-NMS-0040" || attacker.card.name === "黃金鬥士 金牛座" || attacker.card.name?.includes("黃金鬥士")) && !window.isUnitSilenced(attacker, attZone, attIdx);
  const defenderIsTaurus40 = defender && defender.card && (defender.card.id === "SSSR-NMS-0040" || defender.card.name === "黃金鬥士 金牛座" || defender.card.name?.includes("黃金鬥士")) && !window.isUnitSilenced(defender, defZone, defIdx);

  if (attackerIsTaurus40 && window.XLW_isShieldUnit(defender)) {
    defPower = getUnitStars(defender, defZone, defIdx);
    logBattle(`✨ 黃金鬥士 金牛座 效果發動：與阻擋單位戰鬥時，防守方【${defender.card.name}】的攻擊力視為其星數 (${defPower})！`);
  }
  if (defenderIsTaurus40 && window.XLW_isShieldUnit(attacker)) {
    atkPower = getUnitStars(attacker, attZone, attIdx);
    logBattle(`✨ 黃金鬥士 金牛座 效果發動：與阻擋單位戰鬥時，進攻方【${attacker.card.name}】的攻擊力視為其星數 (${atkPower})！`);
  }

  // 1. 播放攻擊力對決 VS 動畫 (等待完畢)
  await showCombatNumberAnimation(attZone, attIdx, defZone, defIdx, atkPower, defPower);

  // 2. 阻擋/盾牌防禦判定
  if (window.XLW_isShieldUnit(attacker)) {
    attacker.attacking = false;
    attacker.target = null;
    return `${unitName(attacker)} 是阻擋/盾牌單位，無法發起有效進攻`;
  }
  if (window.XLW_isShieldUnit(defender)) {
    if (hasArmorBreak(attacker, attZone, attIdx)) {
      logBattle(`✨ 破甲效果：進攻單位 ${attacker.card.name} 具有破甲，無視防守單位的阻擋，繼續交戰！`);
    } else {
      attacker.tapped = true;
      attacker.attacking = false;
      attacker.target = null;
      return `${unitName(defender)} 是阻擋/盾牌單位，本次進攻被成功阻擋`;
    }
  }

  // 3. 讀取雙方遠程、劇毒、偷襲特性
  const attackerCard = attacker.card || attacker;
  const defenderCard = defender.card || defender;

  const attackerHasRanged = (attackerCard.keywords && attackerCard.keywords.includes("\u9060\u7a0b")) ||
                             (attackerCard.effect_text && attackerCard.effect_text.includes("\u9060\u7a0b")) ||
                             attacker.hasRanged;
  const attackerHasPoison = window.xlwUnitHasPoison(attacker, attZone, attIdx);
  const attackerHasSneak = window.xlwUnitHasSneak(attacker, attZone);

  const defenderHasPoison = window.xlwUnitHasPoison(defender, defZone, defIdx);

  // 4. 計算雙方是否應當被擊破 (引入 Ranged 免疫反擊與 Poisonous 必殺邏輯)
  let attackerShouldDie = false;
  let defenderShouldDie = false;

  // 基礎戰鬥數值對決 (相同攻擊力時攻擊方獲勝，防守方被破壞)
  if (atkPower > defPower) {
    defenderShouldDie = true;
    attackerShouldDie = false;
  } else if (defPower > atkPower) {
    attackerShouldDie = true;
    defenderShouldDie = false;
  } else {
    // 平手 (atkPower === defPower)
    // AI 大三元 (SR-FMS-0016)
  const tripleIdx = hand.findIndex(c => c && (c.id === "SR-FMS-0016" || c.name?.includes("大三元")));
  if (tripleIdx >= 0 && window.xlwCheckMahjongCombo(false, "大三元")) {
    const spellCard = hand.splice(tripleIdx, 1)[0];
    await xlwShowSpellActivationOverlay(spellCard, "enemy");
    await aiCastSpell(spellCard, async () => {
      enemyBonusScore += 8;
      enemyExileZone.push(spellCard);
      logBattle("✨ 對手 AI 大三元 效果發動：對手成功胡牌，額外分數 +8★！此卡被除外。");
      renderScore();
      render();
    });
  }

  // AI 小四喜 (R-FMS-0017)
  const fourIdx = hand.findIndex(c => c && (c.id === "R-FMS-0017" || c.name?.includes("小四喜")));
  if (fourIdx >= 0 && window.xlwCheckMahjongCombo(false, "小四喜")) {
    const spellCard = hand.splice(fourIdx, 1)[0];
    await xlwShowSpellActivationOverlay(spellCard, "enemy");
    await aiCastSpell(spellCard, async () => {
      enemyBonusScore += 8;
      enemyExileZone.push(spellCard);
      logBattle("✨ 對手 AI 小四喜 效果發動：對手成功胡牌，額外分數 +8★！此卡被除外。");
      renderScore();
      render();
    });
  }

  // 天下第一獸人作弊大會 (ORC-0019) 效果發動：平手視為我方獲勝！
    let playerCheated = false;
    let enemyCheated = false;
    
    if (defender.card && (defender.card.faction === "獸人" || defender.card.id?.includes("ORC")) && !window.isUnitSilenced(defender, defZone, defIdx)) {
      if (defZone.startsWith("player_") && window.XLW_orcCheatFairActive) {
        playerCheated = true;
      } else if (defZone.startsWith("enemy_") && window.XLW_enemyOrcCheatFairActive) {
        enemyCheated = true;
      }
    }
    if (attacker.card && (attacker.card.faction === "獸人" || attacker.card.id?.includes("ORC")) && !window.isUnitSilenced(attacker, attZone, attIdx)) {
      if (attZone.startsWith("player_") && window.XLW_orcCheatFairActive) {
        playerCheated = true;
      } else if (attZone.startsWith("enemy_") && window.XLW_enemyOrcCheatFairActive) {
        enemyCheated = true;
      }
    }

    if (playerCheated && !enemyCheated) {
      if (attZone.startsWith("player_")) {
        defenderShouldDie = true;
        attackerShouldDie = false;
      } else {
        defenderShouldDie = false;
        attackerShouldDie = true;
      }
      logBattle("✨ 天下第一獸人作弊大會 效果：攻擊力平手改為我方戰鬥成功！");
    } else if (enemyCheated && !playerCheated) {
      if (attZone.startsWith("enemy_")) {
        defenderShouldDie = true;
        attackerShouldDie = false;
      } else {
        defenderShouldDie = false;
        attackerShouldDie = true;
      }
      logBattle("✨ 對手 天下第一獸人作弊大會 效果：攻擊力平手改為對手戰鬥成功！");
    } else {
      // 預設規則：攻擊方獲勝，防守方被破壞
      defenderShouldDie = true;
      attackerShouldDie = false;
    }
  }

  // 狂野巨獸 金牛座 (SSSR-NMS-0044) 戰鬥判定覆寫
  const attackerIsTaurus = (attackerCard.id === "SSSR-NMS-0044" || attackerCard.name?.includes("金牛座")) && !window.isUnitSilenced(attacker, attZone, attIdx);
  const defenderIsTaurus = (defenderCard.id === "SSSR-NMS-0044" || defenderCard.name?.includes("金牛座")) && !window.isUnitSilenced(defender, defZone, defIdx);

  if (attackerIsTaurus) {
    const attackerStars = getUnitStars(attacker, attZone, attIdx);
    const defenderStars = getUnitStars(defender, defZone, defIdx);
    if (defenderStars <= attackerStars) {
      defenderShouldDie = true;
      attackerShouldDie = false;
      logBattle(`✨ 狂野巨獸 金牛座 效果發動：對手單位星數 (${defenderStars}) <= 金牛座星數 (${attackerStars})，金牛座免於戰鬥破壞，且對手單位必定會被戰鬥破壞！`);
    }
  } else if (defenderIsTaurus) {
    const attackerStars = getUnitStars(attacker, attZone, attIdx);
    const defenderStars = getUnitStars(defender, defZone, defIdx);
    if (attackerStars <= defenderStars) {
      attackerShouldDie = true;
      defenderShouldDie = false;
      logBattle(`✨ 狂野巨獸 金牛座 效果發動：攻擊方單位星數 (${attackerStars}) <= 金牛座星數 (${defenderStars})，金牛座免於戰鬥破壞，且攻擊方單位必定會被戰鬥破壞！`);
    }
  }

  // 遠程 (Ranged) 效果應用：進攻時不會因為戰鬥失敗而被破壞（免除反擊傷害）
  if (attackerHasRanged) {
    attackerShouldDie = false;
  }

  // 拆卸 (Disassemble) 效果應用：進攻合體單位時無視數值直接判定戰鬥成功且破壞對方
  const attackerHasDisassemble = (attackerCard.keywords && attackerCard.keywords.includes("拆卸")) ||
                                 (attackerCard.effect_text && attackerCard.effect_text.includes("拆卸")) ||
                                 attackerCard.id === "NMS-0013" || attackerCard.name?.includes("笨手笨腳的維修員");
  const defenderIsCombineUnit = window.isCombineUnit(defender);
  if (attackerHasDisassemble && defenderIsCombineUnit && !window.isUnitSilenced(attacker, attZone, attIdx)) {
    defenderShouldDie = true;
    attackerShouldDie = false;
    logBattle(`✨ 笨手笨腳的維修員 效果：【${attackerCard.name}】發動拆卸效果，進攻對手合體單位【${defenderCard.name}】，直接判定戰鬥成功並將其破壞！`);
  }

  // 劇毒 (Poisonous) 效果應用：與此單位戰鬥的單位將直接被破壞（對遠程單位的進攻無效）
  if (attackerHasPoison) {
    defenderShouldDie = true; // 劇毒進攻必殺
  }
  if (defenderHasPoison) {
    if (!attackerHasRanged) {
      attackerShouldDie = true; // 劇毒防守觸殺 (遠程進攻免疫此效果)
    }
  }

  // 5. 執行擊破與死鬥送墓
  let piercingMsg = "";
  const destPromise = [];

  const attackerIsKnightCat = (attackerCard.id === "CAT-0015" || attackerCard.id === "CAT_0015" || attackerCard.name.includes("騎士喵") ||
                               attackerCard.id === "NEU-0022" || attackerCard.id === "SSSR-NMS-0022" || attackerCard.name.includes("封印騎士"));
  const defenderIsKnightCat = (defenderCard.id === "CAT-0015" || defenderCard.id === "CAT_0015" || defenderCard.name.includes("騎士喵") ||
                               defenderCard.id === "NEU-0022" || defenderCard.id === "SSSR-NMS-0022" || defenderCard.name.includes("封印騎士"));

  // 漏食球 (SR-CAT-0053) 被動效果
  if (defenderShouldDie && (defenderCard.id === "SR-CAT-0053" || defenderCard.id === "CAT_053" || defenderCard.name.includes("漏食球"))) {
    if (!window.XLW_foodBallTriggeredTurn || window.XLW_foodBallTriggeredTurn !== turn) {
      if (defenderOwner === "enemy") {
        // 我方為進攻方，對手場上有漏食球被我方進攻，我方（進攻方）決定是否發動效果
        const hasEmpty = ["player_front", "player_back"].some(z => field[z].some(u => u === null));
        if (hasEmpty) {
          const choice = await showXLWConfirm("漏食球 效果發動", "是否召喚 1 個小旅人到我方場上以代替破壞對手的 漏食球？");
          if (isMultiplayer) {
            ws.send(JSON.stringify({
              action: "foodball_choice",
              zone: defZone,
              idx: defIdx,
              trigger: choice
            }));
          }
          if (choice) {
            defenderShouldDie = false;
            window.XLW_foodBallTriggeredTurn = turn;
            
            // 手動放置小旅人到我方場上
            window.XLW_currentFoodBallZone = defZone;
            window.XLW_currentFoodBallIdx = defIdx;
            window.XLW_isFoodBallSummoning = true;
            window.XLW_bypassNormalSummonLimit = true;
            window.XLW_bypassTurnCheck = true;
            selectedHandForSummon = "TRAVELER";
            setStatus("【漏食球 召喚】請選擇我方一個空格進行特殊召喚！");
            render();
            
            await new Promise(r => { window.XLW_summonPlacementResolve = r; });
            
            window.XLW_bypassTurnCheck = false;
            window.XLW_isFoodBallSummoning = false;
          }
        }
      } else {
        // 我方為防守方，對手(AI/敵方玩家)為進攻方，對手決定是否發動效果
        const hasEmpty = ["enemy_front", "enemy_back"].some(z => field[z].some(u => u === null));
        if (hasEmpty) {
          if (isMultiplayer) {
            // 聯網同步：等待對手傳送其漏食球效果發動決定！
            const key = `${defZone}:${defIdx}`;
            if (window.XLW_opponentFoodBallChoice[key] === undefined) {
              await new Promise(resolve => {
                const check = setInterval(() => {
                  if (window.XLW_opponentFoodBallChoice[key] !== undefined) {
                    clearInterval(check);
                    resolve();
                  }
                }, 50);
              });
            }
            const opponentTriggered = window.XLW_opponentFoodBallChoice[key];
            if (opponentTriggered) {
              defenderShouldDie = false;
              window.XLW_foodBallTriggeredTurn = turn;
              // 等對手傳送其放置位置
              if (window.XLW_opponentFoodBallSummon[key] === undefined) {
                await new Promise(resolve => {
                  const check = setInterval(() => {
                    if (window.XLW_opponentFoodBallSummon[key] !== undefined) {
                      clearInterval(check);
                      resolve();
                    }
                  }, 50);
                });
              }
              const placement = window.XLW_opponentFoodBallSummon[key];
              field[placement.zone][placement.idx] = {
                card: structuredClone(LITTLE_TRAVELER),
                tapped: false,
                attacking: false,
                target: null,
                summonedTurn: turn,
                summonedZone: placement.zone
              };
              logBattle(`漏食球 效果：對手召喚 1 個小旅人代替破壞 漏食球。`);
            }
          } else {
            // 單機 AI 模式：AI為進攻方，50% 機率發動並召喚到AI場上（我方的enemy_front/back）
            if (Math.random() < 0.5) {
              defenderShouldDie = false;
              window.XLW_foodBallTriggeredTurn = turn;
              let placed = false;
              for (const z of ["enemy_front", "enemy_back"]) {
                for (let i = 0; i < 5; i++) {
                  if (!field[z][i]) {
                    field[z][i] = {
                      card: structuredClone(LITTLE_TRAVELER),
                      tapped: false,
                      attacking: false,
                      target: null,
                      summonedTurn: turn,
                      summonedZone: z
                    };
                    logBattle(`漏食球 效果：對手召喚 1 個小旅人代替破壞 漏食球。`);
                    placed = true;
                    break;
                  }
                }
                if (placed) break;
              }
            }
          }
        }
      }
    }
  }

  // --- 二連爪擊喵 (SR-CAT-0047) 戰鬥破壞無效與攻擊力 +3 效果 ---
  if (attackerShouldDie) {
    const isDoubleClawAttacking = (attackerCard.id === "SR-CAT-0047" || attackerCard.name === "二連爪擊喵");
    if (isDoubleClawAttacking) {
      if (defenderOwner === "enemy") {
        if (!attacker.doubleClawUsed) {
          const confirmNegate = await showXLWConfirm(
            "二連爪擊喵 效果發動",
            "【二連爪擊喵】即將被戰鬥破壞！是否發動效果無效化本次破壞，並獲得 +3 攻擊力直到防守階段結束？"
          );
          if (isMultiplayer) {
            ws.send(JSON.stringify({
              action: "double_claw_choice",
              attZone: attZone,
              attIdx: attIdx,
              negate: confirmNegate
            }));
          }
          if (confirmNegate) {
            attackerShouldDie = false;
            attacker.doubleClawUsed = true;
            attacker.doubleClawBonus = true;
            attacker.atkModifier = (attacker.atkModifier || 0) + 3;
            logBattle(`✨ 效果發動：我方【二連爪擊喵】無效化了本次戰鬥破壞，並獲得 +3 攻擊力！`);
            render();
          }
        }
      } else {
        if (isMultiplayer) {
          const key = `${attZone}:${attIdx}`;
          if (window.XLW_opponentDoubleClawNegate[key] === undefined) {
            await new Promise(resolve => {
              const check = setInterval(() => {
                if (window.XLW_opponentDoubleClawNegate[key] !== undefined) {
                  clearInterval(check);
                  resolve();
                }
              }, 50);
            });
          }
          const opponentNegated = window.XLW_opponentDoubleClawNegate[key];
          if (opponentNegated) {
            attackerShouldDie = false;
            attacker.doubleClawBonus = true;
            attacker.atkModifier = (attacker.atkModifier || 0) + 3;
            logBattle(`✨ 效果發動：對手的【二連爪擊喵】無效化了本次戰鬥破壞，並獲得 +3 攻擊力！`);
            render();
          }
        } else {
          if (!attacker.doubleClawUsed) {
            attackerShouldDie = false;
            attacker.doubleClawUsed = true;
            attacker.doubleClawBonus = true;
            attacker.atkModifier = (attacker.atkModifier || 0) + 3;
            logBattle(`✨ 效果發動：對手AI的【二連爪擊喵】無效化了本次戰鬥破壞，並獲得 +3 攻擊力！`);
            render();
          }
        }
      }
    }
  }

  // 永鬥神 奈祖爾 combat shield (attacker side)
  if (attackerShouldDie && attacker && attacker.card && (attacker.card.id === "SR-ORC-0026" || attacker.card.id === "SSR-ORC-0026" || attacker.card.name?.includes("奈祖爾")) && !window.isUnitSilenced(attacker, attZone, attIdx)) {
    const isPlayer = attZone.startsWith("player_");
    const bonusScore = isPlayer ? playerBonusScore : enemyBonusScore;
    if (bonusScore > 0) {
      let confirmShield = true;
      if (isPlayer) {
        confirmShield = await showXLWConfirm("永鬥神 奈祖爾 效果發動", "偵測到永鬥神將被破壞，是否消耗 1 點獎勵分使其免於破壞？");
      }
      if (confirmShield) {
        if (isPlayer) {
          playerBonusScore -= 1;
        } else {
          enemyBonusScore -= 1;
        }
        attackerShouldDie = false;
        logBattle(`✨ 永鬥神 奈祖爾 效果：消耗 1 點獎勵分，免除本次戰鬥破壞！`);
        renderScore();
      }
    }
  }
  // 永鬥神 奈祖爾 combat shield (defender side)
  if (defenderShouldDie && defender && defender.card && (defender.card.id === "SR-ORC-0026" || defender.card.id === "SSR-ORC-0026" || defender.card.name?.includes("奈祖爾")) && !window.isUnitSilenced(defender, defZone, defIdx)) {
    const isPlayer = defZone.startsWith("player_");
    const bonusScore = isPlayer ? playerBonusScore : enemyBonusScore;
    if (bonusScore > 0) {
      let confirmShield = true;
      if (isPlayer) {
        confirmShield = await showXLWConfirm("永鬥神 奈祖爾 效果發動", "偵測到永鬥神將被破壞，是否消耗 1 點獎勵分使其免於破壞？");
      }
      if (confirmShield) {
        if (isPlayer) {
          playerBonusScore -= 1;
        } else {
          enemyBonusScore -= 1;
        }
        defenderShouldDie = false;
        logBattle(`✨ 永鬥神 奈祖爾 效果：消耗 1 點獎勵分，免除本次戰鬥破壞！`);
        renderScore();
      }
    }
  }

  if (attackerShouldDie && isUnitIndestructible(attacker, attZone, attIdx, true)) {
    attackerShouldDie = false;
    logBattle(`✨ ${unitName(attacker)} 具有破壞抗性，免於被戰鬥破壞！`);
  }
  if (defenderShouldDie && isUnitIndestructible(defender, defZone, defIdx, true)) {
    defenderShouldDie = false;
    logBattle(`✨ ${unitName(defender)} 具有破壞抗性，免於被戰鬥破壞！`);
  }

  // 💊 復活膠囊與射手座戰鬥效果判定
  if (defenderOwner === "enemy" && attackerShouldDie) {
    const capsuleCard = hand.find(c => c && (c.id === "R-NMG-0032" || c.name?.includes("復活膠囊")));
    if (capsuleCard) {
      const confirmUse = await showXLWConfirm(
        "💊 復活膠囊 效果發動",
        `我方進攻單位 <b>${attacker.card.name}</b> 即將在戰鬥中被破壞，是否從手牌中發動【復活膠囊】使其免於破壞並保留在場上？`
      );
      if (confirmUse) {
        hand.splice(hand.indexOf(capsuleCard), 1);
        graveyard.push(capsuleCard);
        attackerShouldDie = false;
        logBattle(`✨ 復活膠囊 效果發動：免除我方 ${attacker.card.name} 的戰鬥破壞！`);
        if (isMultiplayer) {
          sendFullGameStateToOpponent();
        }
      }
    }
  }
  if (defenderOwner === "player" && attackerShouldDie && !isMultiplayer) {
    const enemyHand = window.XLW_ENEMY?.hand || [];
    const capsuleCard = enemyHand.find(c => c && (c.id === "R-NMG-0032" || c.name?.includes("復活膠囊")));
    if (capsuleCard && Math.random() < 0.5) {
      enemyHand.splice(enemyHand.indexOf(capsuleCard), 1);
      window.XLW_ENEMY.grave.push(capsuleCard);
      attackerShouldDie = false;
      logBattle(`✨ 對手 AI 發動【復活膠囊】：免除對手 ${attacker.card.name} 的戰鬥破壞！`);
    }

  let defenderKidnapped = false;
  if (defenderShouldDie && !attackerShouldDie) {
    const isSagittarius = attacker.card && (attacker.card.id === "SSSR-NMS-0046" || attacker.card.name.includes("奪星射手") || attacker.card.name.includes("射手座"));
    if (isSagittarius) {
      const isDefenderTraveler = defender.card.name.includes("小旅人") || defender.card.name.includes("尋寶小旅人") || defender.card.id === "TOKEN_TRAVELER" || defender.card.id === "NEU-0010";
      if (!isDefenderTraveler) {
        defenderShouldDie = false;
        defenderKidnapped = true;
        await kidnapCard(defZone, defIdx, attZone, attIdx);
      }
    }
  }

  // 手賤喵 (C-CAT-0054) 效果
  let handyMeowTriggered = false;
  let defeatedUnit = null;
  let defeatedZone = null;
  let defeatedIdx = -1;
  let victorUnit = null;
  let victorZone = null;
  let victorIdx = -1;
  let victorOwner = null;

  if (defenderShouldDie && !attackerShouldDie && !defenderKidnapped) {
    if (attacker && attacker.card && (attacker.card.id === "C-CAT-0054" || attacker.card.name?.includes("手賤喵")) && !window.isUnitSilenced(attacker, attZone, attIdx)) {
      handyMeowTriggered = true;
      defeatedUnit = defender;
      defeatedZone = defZone;
      defeatedIdx = defIdx;
      victorUnit = attacker;
      victorZone = attZone;
      victorIdx = attIdx;
      victorOwner = defenderOwner === "player" ? "enemy" : "player";
    }
  } else if (attackerShouldDie && !defenderShouldDie) {
    if (defender && defender.card && (defender.card.id === "C-CAT-0054" || defender.card.name?.includes("手賤喵")) && !window.isUnitSilenced(defender, defZone, defIdx)) {
      handyMeowTriggered = true;
      defeatedUnit = attacker;
      defeatedZone = attZone;
      defeatedIdx = attIdx;
      victorUnit = defender;
      victorZone = defZone;
      victorIdx = defIdx;
      victorOwner = defenderOwner;
    }
  }

  if (handyMeowTriggered) {
    const getEmptySlotsForSide = (owner) => {
      const prefix = owner === "player" ? "player_" : "enemy_";
      const slots = [];
      for (const z of [prefix + "front", prefix + "back"]) {
        for (let i = 0; i < 5; i++) {
          if (!field[z][i]) {
            slots.push({ zone: z, idx: i });
          }
        }
      }
      return slots;
    };

    const loserOwner = victorOwner === "player" ? "enemy" : "player";
    const emptySlots = getEmptySlotsForSide(loserOwner);
    if (emptySlots.length > 0) {
      let chosenSlot = null;
      if (victorOwner === "player") {
        const choices = emptySlots.map((s, i) => ({
          text: `${s.zone.includes("front") ? "對手前排" : "對手後排"}${s.idx + 1}`,
          value: i
        }));
        const chosenIdx = await showXLWChoiceModal("【手賤喵】效果發動", "請選擇要將被擊破的敵方單位移動至哪一個空格：", choices);
        if (chosenIdx !== null && chosenIdx !== undefined) {
          chosenSlot = emptySlots[chosenIdx];
        }
        if (isMultiplayer && ws && chosenSlot) {
          ws.send(JSON.stringify({
            action: "handy_meow_move_sync",
            fromZone: defeatedZone,
            fromIdx: defeatedIdx,
            toZone: chosenSlot.zone,
            toIdx: chosenSlot.idx
          }));
        }
      } else {
        if (isMultiplayer) {
          const key = `${defeatedZone}:${defeatedIdx}`;
          if (window.XLW_handyMeowOpponentChoice[key] === undefined) {
            await new Promise(resolve => {
              const check = setInterval(() => {
                if (window.XLW_handyMeowOpponentChoice[key] !== undefined) {
                  clearInterval(check);
                  resolve();
                }
              }, 50);
            });
          }
          chosenSlot = window.XLW_handyMeowOpponentChoice[key];
        } else {
          chosenSlot = emptySlots[Math.floor(Math.random() * emptySlots.length)];
        }
      }

      if (chosenSlot) {
        logBattle(`✨ 手賤喵 效果：將原本要擊破的【${defeatedUnit.card.name}】移動到 ${loserOwner === "player" ? "我方" : "對手"}${chosenSlot.zone.includes("front") ? "前排" : "後排"}${chosenSlot.idx + 1}！`);
        
        field[chosenSlot.zone][chosenSlot.idx] = {
          card: defeatedUnit.card,
          tapped: defeatedUnit.tapped,
          attacking: false,
          target: null,
          summonedTurn: defeatedUnit.summonedTurn,
          summonedZone: chosenSlot.zone
        };
        field[defeatedZone][defeatedIdx] = null;
        
        if (defenderShouldDie && !attackerShouldDie) {
          defenderShouldDie = false;
        } else {
          attackerShouldDie = false;
        }
        render();
      }
    }
  }

  if (attackerShouldDie) {
    destPromise.push(destroyUnit(attZone, attIdx, defenderOwner === "player" ? "enemy" : "player", defenderIsKnightCat, true));
  }
  if (defenderShouldDie) {
    destPromise.push(destroyUnit(defZone, defIdx, defenderOwner, attackerIsKnightCat, true));

    // 貫穿 (Piercing) 效果判定：擊破前排後繼續進攻後排
    const hasPiercing = checkUnitHasPiercing(attacker, attZone, attIdx);
    if (hasPiercing && defZone.includes("front")) {
      const backZone = defZone.replace("front", "back");
      const backUnit = field[backZone]?.[defIdx];
      if (backUnit) {
        destPromise.push(destroyUnit(backZone, defIdx, defenderOwner, false, true));
        piercingMsg = `，並貫穿擊破了後排的 ${unitName(backUnit)}`;
      }
    }
  }

  if (destPromise.length > 0) {
    await Promise.all(destPromise);
    
    // --- 狂歡跳跳蛙 (SSSR-NMS-0049) 效果 ---
    const playerUnitDestroyedInCombat = (defenderShouldDie && defenderOwner === "player") || (attackerShouldDie && defenderOwner === "enemy");
    const destroyedPlayerZone = (defenderShouldDie && defenderOwner === "player") ? defZone : ((attackerShouldDie && defenderOwner === "enemy") ? attZone : null);
    const destroyedPlayerIdx = (defenderShouldDie && defenderOwner === "player") ? defIdx : ((attackerShouldDie && defenderOwner === "enemy") ? attIdx : -1);
    
    if (playerUnitDestroyedInCombat && destroyedPlayerZone && destroyedPlayerIdx !== -1) {
      let frogPos = null;
      for (const z of ["player_front", "player_back"]) {
        field[z].forEach((u, i) => {
          if (u && u.card && (u.card.id === "SSSR-NMS-0049" || u.card.name?.includes("狂歡跳跳蛙")) && (z !== destroyedPlayerZone || i !== destroyedPlayerIdx) && !window.isUnitSilenced(u, z, i)) {
            frogPos = { zone: z, idx: i, unit: u };
          }
        });
      }
      if (frogPos) {
        const moveFrog = await showXLWConfirm(
          "狂歡跳跳蛙 效果發動",
          `偵測到我方單位被戰鬥破壞！是否將場上的【狂歡跳跳蛙】移動到該單位原本的格子 (${destroyedPlayerZone.includes("front") ? "前排" : "後排"}${destroyedPlayerIdx + 1})？`
        );
        if (moveFrog) {
          field[frogPos.zone][frogPos.idx] = null;
          field[destroyedPlayerZone][destroyedPlayerIdx] = frogPos.unit;
          frogPos.unit.summonedZone = destroyedPlayerZone;
          logBattle(`✨ 狂歡跳跳蛙 效果：從 ${frogPos.zone.includes("front") ? "前排" : "後排"}${frogPos.idx + 1} 移動到被破壞單位的格子 ${destroyedPlayerZone.includes("front") ? "前排" : "後排"}${destroyedPlayerIdx + 1}！`);
          if (isMultiplayer && ws) {
            ws.send(JSON.stringify({
              action: "frog_move_sync",
              fromZone: frogPos.zone,
              fromIdx: frogPos.idx,
              toZone: destroyedPlayerZone,
              toIdx: destroyedPlayerIdx
            }));
          }
          render();
        }
      }
    }
    
    const enemyUnitDestroyedInCombat = (defenderShouldDie && defenderOwner === "enemy") || (attackerShouldDie && defenderOwner === "player");
    const destroyedEnemyZone = (defenderShouldDie && defenderOwner === "enemy") ? defZone : ((attackerShouldDie && defenderOwner === "player") ? attZone : null);
    const destroyedEnemyIdx = (defenderShouldDie && defenderOwner === "enemy") ? defIdx : ((attackerShouldDie && defenderOwner === "player") ? attIdx : -1);
    
    if (enemyUnitDestroyedInCombat && destroyedEnemyZone && destroyedEnemyIdx !== -1 && !isMultiplayer) {
      let aiFrogPos = null;
      for (const z of ["enemy_front", "enemy_back"]) {
        field[z].forEach((u, i) => {
          if (u && u.card && (u.card.id === "SSSR-NMS-0049" || u.card.name?.includes("狂歡跳跳蛙")) && (z !== destroyedEnemyZone || i !== destroyedEnemyIdx) && !window.isUnitSilenced(u, z, i)) {
            aiFrogPos = { zone: z, idx: i, unit: u };
          }
        });
      }
      if (aiFrogPos) {
        field[aiFrogPos.zone][aiFrogPos.idx] = null;
        field[destroyedEnemyZone][destroyedEnemyIdx] = aiFrogPos.unit;
        aiFrogPos.unit.summonedZone = destroyedEnemyZone;
        logBattle(`✨ 狂歡跳跳蛙 效果：對手將場上的【狂歡跳跳蛙】移動到被破壞單位的格子 ${destroyedEnemyZone.includes("front") ? "前排" : "後排"}${destroyedEnemyIdx + 1}！`);
        render();
      }
    }
  }

  // 統計與套用戰鬥成功獎勵
  const attackerSuccess = (!attackerShouldDie && defenderShouldDie);
  const defenderSuccess = (!defenderShouldDie && attackerShouldDie);
  if (attackerSuccess) {
    const successOwner = (defenderOwner === "enemy" ? "player" : "enemy");
    await applyCombatSuccessReward(attacker, defender, successOwner, true);

    const isWildBoar = attackerCard.id === "R-NMS-0062" || attackerCard.id === "R-NMS-0062-2" || attackerCard.name?.includes("野豬");
    if (isWildBoar) {
      logBattle(`✨ 野豬 效果：進攻戰鬥成功，野豬一同送入墓地破壞！`);
      await destroyUnit(attZone, attIdx, successOwner, false, true);
    }
    const isBombOrc = attackerCard.id === "R-ORC-0024" || attackerCard.name?.includes("爆破獸人");
    if (isBombOrc) {
      logBattle(`✨ 爆破獸人 效果：戰鬥成功，自爆破壞！`);
      await destroyUnit(attZone, attIdx, successOwner, false, true);
    }
    // 懸賞獸人 (R-ORC-0059) 被動
    const hasBountyOrc = ["player_front", "player_back", "enemy_front", "enemy_back"].some(z => {
      const isPlayer = z.startsWith("player_");
      const matchOwner = isPlayer ? "player" : "enemy";
      if (matchOwner !== successOwner) return false;
      return field[z].some((u, i) => u && u.card && (u.card.id === "R-ORC-0059" || u.card.name?.includes("懸賞獸人")) && !window.isUnitSilenced(u, z, i));
    });
    if (hasBountyOrc) {
      const extraDeck = successOwner === "player" ? window.XLW_playerExtraDeck : (window.XLW_enemyExtraDeck || []);
      const targetIdx = extraDeck.findIndex(c => c && (c.id === "R-ORC-0058" || c.name?.includes("懸賞令")));
      if (targetIdx >= 0) {
        const bountyCard = extraDeck.splice(targetIdx, 1)[0];
        if (successOwner === "player") {
          hand.push(bountyCard);
          logBattle(`✨ 懸賞獸人 被動效果：由於獸人戰鬥成功，從額外區將【懸賞令】加入手牌！`);
        } else {
          if (!window.XLW_ENEMY.hand) window.XLW_ENEMY.hand = [];
          window.XLW_ENEMY.hand.push(bountyCard);
          logBattle(`✨ 對手 懸賞獸人 被動效果：由於對手獸人戰鬥成功，從其額外區將【懸賞令】加入手牌！`);
        }
        render();
      }
    }
    if (false) {
      logBattle(`✨ 野豬 效果：進攻戰鬥成功，野豬一同送入墓地破壞！`);
      await destroyUnit(attZone, attIdx, successOwner, false, true);
    }

    if (successOwner === "player") {
      const isMeow = attacker.card && (attacker.card.deck === "喵喵賊" || attacker.card.faction === "喵喵賊" || (attacker.card.id && (String(attacker.card.id).includes("CAT") || String(attacker.card.id).includes("cat"))));
      if (isMeow) {
        window.XLW_turnCombatWins = (window.XLW_turnCombatWins || 0) + 1;
        logDebug(`【戰鬥成功追蹤】我方喵喵賊單位進攻戰鬥成功！本回合累計成功 ${window.XLW_turnCombatWins} 次。`);
        await xlwCheckDoubleClawMeowUpgrade();
      }
    }

    // --- 血戰幫狼牙棒 額外啟動效果 ---
    if (successOwner === "player") {
      const isOrc = attacker.card && (attacker.card.faction === "獸人" || attacker.card.race === "獸人" || attacker.card.id.includes("ORC") || attacker.card.id.includes("0RC"));
      const hasOtherEquip = attacker.equipments && attacker.equipments.some(eqName => {
        if (eqName === "血戰幫狼牙棒") return false;
        if (attacker.tempEquips && attacker.tempEquips[eqName] !== undefined) return false;
        if (["削弱藥水", "振奮藥水", "塗毒", "睡眠反擊拳"].some(x => eqName.includes(x))) return false;
        return true;
      });
      const maceInExtra = playerExtraDeck.find(c => c && (c.id === "R-ORC-0054" || c.name === "血戰幫狼牙棒"));

      if (isOrc && hasOtherEquip && maceInExtra) {
        const confirmAdd = await showXLWConfirm(
          "狼牙棒 效果觸發",
          `我方獸人單位 ${attacker.card.name} 進攻成功且裝備了其他裝備卡！是否將【血戰幫狼牙棒】自額外牌庫加入手牌？`,
          "是",
          "否"
        );
        if (confirmAdd) {
          playerExtraDeck.splice(playerExtraDeck.indexOf(maceInExtra), 1);
          hand.push(maceInExtra);
          logBattle(`✨ 血戰幫狼牙棒 效果：將【血戰幫狼牙棒】自額外牌庫加入我方手牌！`);
          
          if (isMultiplayer) {
            ws.send(JSON.stringify({
              action: "add_to_hand_sync"
            }));
          }
          render();
        }
      }
    }
    // ---------------------------------
  }
  if (defenderSuccess) {
    const successOwner = defenderOwner;
    await applyCombatSuccessReward(defender, attacker, successOwner, false);
  }

  // 6. 偷襲 (Sneak Attack) 效果應用：成功進攻並存活後，將此卡收回手牌
  let sneakMsg = "";
  if (attackerHasSneak && !attackerShouldDie && defenderShouldDie) {
    if (defenderOwner === "player") {
      // 攻擊方是對手
      if (isMultiplayer) {
        // 聯網同步：等待對手（進攻方）傳送其偷襲回手決定！
        const key = `${attZone}:${attIdx}`;
        if (window.XLW_opponentSneakRecall[key] === undefined) {
          await new Promise(resolve => {
            const check = setInterval(() => {
              if (window.XLW_opponentSneakRecall[key] !== undefined) {
                clearInterval(check);
                resolve();
              }
            }, 50);
          });
        }
        const opponentRecalled = window.XLW_opponentSneakRecall[key];
        if (opponentRecalled) {
          if (attacker.equipments && attacker.equipments.length > 0) {
            attacker.equipments.forEach(eqName => {
              const isTempMagic = ["削弱藥水", "振奮藥水", "塗毒", "睡眠反擊拳"].some(x => eqName.includes(x));
              if (isTempMagic) return;
              const cardObj = allCards.find(c => c && c.name === eqName) || { name: eqName, type: "magic" };
              const cloned = structuredClone(cardObj);
              cloned.type = "magic";
              window.XLW_ENEMY.grave.push(cloned);
              enemyGraveyard = window.XLW_ENEMY.grave;
              logBattle(`✨ 裝備卡 ${cloned.name} 因對手單位回手進入對手墓地。`);
            });
            attacker.equipments = [];
          }
          if (window.XLW_ENEMY.hand) {
            window.XLW_ENEMY.hand.push(attacker.card);
          }
          await cleanUnitLeavingField(attacker, attZone);
          field[attZone][attIdx] = null;
          sneakMsg = `，且對手觸發「偷襲」成功收回其手牌`;
          await triggerEnemySneakAttackSuccessEffects(attacker.card);
        } else {
          attacker.tapped = true;
          attacker.attacking = false;
          attacker.target = null;
          sneakMsg = `，對手選擇放棄「偷襲」回手，使該單位留在場上處於疲勞狀態`;
        }
      } else {
        if (attacker.equipments && attacker.equipments.length > 0) {
          attacker.equipments.forEach(eqName => {
            const isTempMagic = ["削弱藥水", "振奮藥水", "塗毒", "睡眠反擊拳"].some(x => eqName.includes(x));
            if (isTempMagic) return;
            const cardObj = allCards.find(c => c && c.name === eqName) || { name: eqName, type: "magic" };
            const cloned = structuredClone(cardObj);
            cloned.type = "magic";
            window.XLW_ENEMY.grave.push(cloned);
            enemyGraveyard = window.XLW_ENEMY.grave;
            logBattle(`✨ 裝備卡 ${cloned.name} 因對手單位回手進入對手墓地。`);
          });
          attacker.equipments = [];
        }
        if (window.XLW_ENEMY.hand) {
          window.XLW_ENEMY.hand.push(attacker.card);
          await cleanUnitLeavingField(attacker, attZone);
          field[attZone][attIdx] = null;
          sneakMsg = `，且對手觸發「偷襲」成功收回其手牌`;
          await triggerEnemySneakAttackSuccessEffects(attacker.card);
        }
      }
    } else {
      // 攻擊方是我方：彈出奢華確認框讓玩家自行選擇是否回手！
      const userWantsRecall = await showXLWConfirm(
        "🥷 偵測到【偷襲】契機！",
        `我方單位 <b>${unitName(attacker)}</b> 進攻成功！<br>是否啟動「偷襲」將其收回手牌並觸發偷襲成功連動效果？`,
        "確認回手 (啟動偷襲)",
        "留在場上 (放棄回手)"
      );
      
      if (isMultiplayer) {
        ws.send(JSON.stringify({
          action: "sneak_recall",
          attZone: attZone,
          attIdx: attIdx,
          recall: userWantsRecall
        }));
      }
      
      if (userWantsRecall) {
        if (attacker.equipments && attacker.equipments.length > 0) {
          attacker.equipments.forEach(eqName => {
            const cardObj = allCards.find(c => c && c.name === eqName) || { name: eqName, type: "magic" };
            const cloned = structuredClone(cardObj);
            cloned.type = "magic";
            graveyard.push(cloned);
            logBattle(`✨ 裝備卡 ${cloned.name} 因我方單位回手進入我方墓地。`);
          });
          attacker.equipments = [];
        }
        hand.push(attacker.card);
        await cleanUnitLeavingField(attacker, attZone);
        field[attZone][attIdx] = null;
        sneakMsg = `，且觸發「偷襲」成功收回我方手牌`;
        
        // 核心：觸發「偷襲成功」的一系列效果！
        await triggerSneakAttackSuccessEffects(attacker.card);
      } else {
        // 放棄回手，留在場上並處於疲勞狀態
        attacker.tapped = true;
        attacker.attacking = false;
        attacker.target = null;
        sneakMsg = `，我方選擇放棄「偷襲」回手，使該單位留在場上處於疲勞狀態`;

        // 負能量喵 (R-CAT-0046) 效果
        const hasNegativeMeow = ["player_front", "player_back"].some(zone => 
          field[zone].some(u => u && (u.card?.id === "R-CAT-0046" || u.card?.name === "負能量喵"))
        );
        const isMeowCard = (c) => c && (c.deck === "喵喵賊" || c.faction === "喵喵賊" || (c.id && (String(c.id).includes("CAT") || String(c.id).includes("cat"))));
        
        if (hasNegativeMeow && isMeowCard(attacker.card)) {
          logBattle(`✨【負能量喵】效果：我方喵喵賊單位 ${attacker.card.name} 進攻成功且未回手，視為偷襲成功 2 次！`);
          await triggerSneakAttackSuccessEffects(attacker.card);
          await triggerSneakAttackSuccessEffects(attacker.card);
        }
      }
    }
  } else {
    // 未回手之存活單位進行橫置 Tap 鎖定
    if (!attackerShouldDie) {
      attacker.tapped = true;
      attacker.attacking = false;
      attacker.target = null;
    }
  }

  // 7. 回傳詳細戰鬥日誌
  let combatMsg = "";
  if (attackerShouldDie && defenderShouldDie) {
    combatMsg = `${unitName(attacker)} 與 ${unitName(defender)} 實力相當或劇毒互噬，同歸於盡`;
  } else if (attackerShouldDie) {
    combatMsg = `${unitName(attacker)} 攻擊失敗，自身被破壞，而 ${unitName(defender)} 完好無損`;
  } else if (defenderShouldDie) {
    combatMsg = `${unitName(attacker)} 成功擊破了 ${unitName(defender)}${piercingMsg}${sneakMsg}`;
  } else {
    combatMsg = `${unitName(attacker)} 進攻了 ${unitName(defender)}，雙方皆安然無恙${sneakMsg}`;
  }

  // 武士繪 (R-VLG-0011) 戰鬥成功回收效果
  if (!attackerShouldDie && defenderShouldDie && defenderOwner === "enemy") {
    if (attacker.card && attacker.card.name.includes("武士繪")) {
      const yokaiInGrave = graveyard.filter(c => c && c.type === "unit" && (c.faction === "妖怪村莊" || c.id?.includes("VLG")));
      if (yokaiInGrave.length > 0) {
        setTimeout(async () => {
          const choices = yokaiInGrave.map((c, i) => ({ text: `${c.name} (${c.id})`, value: i }));
          const chosen = await showXLWChoiceModal("武士繪 戰鬥成功效果", "請選擇一張墓地中的妖怪村莊單位回收至手牌：", choices);
          if (chosen !== null && chosen !== undefined) {
            const cardToRecover = yokaiInGrave[chosen];
            const graveIdx = graveyard.indexOf(cardToRecover);
            if (graveIdx >= 0) {
              graveyard.splice(graveIdx, 1);
              hand.push(cardToRecover);
              logBattle(`武士繪 效果：成功回收墓地中的 ${cardToRecover.name} 至手牌。`);
              render();
            }
          }
        }, 100);
      }
    }
  }

  return combatMsg;
}
